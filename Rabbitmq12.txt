public class CapacityRequest
{
    public string Namespace { get; set; }
    public long TargetOrders { get; set; }
    public int WithinMinutes { get; set; }
    public TimeSpan HistoricalWindow { get; set; }
    public double SafetyMarginPercent { get; set; }
}

public class CapacityRecommendation
{
    public string Namespace { get; set; }
    public long TargetOrders { get; set; }
    public int WithinMinutes { get; set; }
    public TimeSpan HistoricalWindow { get; set; }
    public double SafetyMarginPercent { get; set; }

    // Historical baseline
    public int CurrentPodCount { get; set; }
    public long HistoricalOrders { get; set; }
    public double HistoricalAvgMemoryBytes { get; set; }
    public double HistoricalP95MemoryBytes { get; set; }
    public double HistoricalMaxMemoryBytes { get; set; }

    // Derived per-order cost
    public double MemoryPerOrderBytes { get; set; }
    public double HistoricalOrdersPerPodPerSecond { get; set; }

    // Required (target)
    public double RequiredThroughputPerSecond { get; set; }
    public int PodsFromThroughput { get; set; }
    public double RequiredTotalMemoryBytes { get; set; }
    public int PodsFromMemory { get; set; }

    // Final recommendation
    public int RecommendedPods { get; set; }
    public double RecommendedMemoryPerPodBytes { get; set; }
    public double RecommendedTotalMemoryBytes { get; set; }

    public DateTime AnalyzedAt { get; set; } = DateTime.UtcNow;

    public string GetSummary() => $"""
        Capacity Recommendation for '{Namespace}'
        ─────────────────────────────────────────
        Target:   {TargetOrders:N0} orders within {WithinMinutes} min
                  → {RequiredThroughputPerSecond:F2} orders/sec
        History:  {HistoricalWindow.TotalDays:F1} days, {HistoricalOrders:N0} orders, {CurrentPodCount} pods
                  → {HistoricalOrdersPerPodPerSecond:F4} orders/pod/sec
                  → {FormatBytes(MemoryPerOrderBytes)} memory/order

        Pods from throughput : {PodsFromThroughput}
        Pods from memory     : {PodsFromMemory}
        Safety margin        : {SafetyMarginPercent}%

        ► Recommended pods   : {RecommendedPods}
        ► Memory per pod     : {FormatBytes(RecommendedMemoryPerPodBytes)}
        ► Total memory       : {FormatBytes(RecommendedTotalMemoryBytes)}
        """;

    private static string FormatBytes(double b)
    {
        string[] s = { "B", "KB", "MB", "GB", "TB" };
        int i = 0;
        while (b >= 1024 && i < s.Length - 1) { b /= 1024; i++; }
        return $"{b:F2} {s[i]}";
    }
}







public async Task<CapacityRecommendation> CalculateRequirementsAsync(
    CapacityRequest request,
    CancellationToken ct = default)
{
    if (request.TargetOrders <= 0) throw new ArgumentException("targetOrders must be > 0");
    if (request.WithinMinutes <= 0) throw new ArgumentException("withinMinutes must be > 0");
    if (request.HistoricalWindow <= TimeSpan.Zero) throw new ArgumentException("historicalWindow must be > 0");

    // Fix the time boundary once so DB and Prometheus see the same window
    var toUtc = DateTime.UtcNow;
    var fromUtc = toUtc - request.HistoricalWindow;

    // 1. Historical memory (Prometheus)
    var memory = await GetHistoricalMemoryUsageAsync(
        request.Namespace, fromUtc, toUtc);
    if (memory == null || !memory.Values.Any())
        throw new InvalidOperationException(
            $"No historical memory data for '{request.Namespace}'.");

    var memValues = memory.Values.Select(v => v.Value).ToArray();
    var avgMem   = memValues.Average();
    var p95Mem   = CalculatePercentile(memValues, 95);
    var maxMem   = memValues.Max();

    // 2. Historical orders (DB) — same window
    var historicalOrders = await _orderRepository.GetOrderCountAsync(
        request.Namespace, fromUtc, toUtc, ct);

    if (historicalOrders <= 0)
        throw new InvalidOperationException(
            $"No historical orders for '{request.Namespace}' in the last {request.HistoricalWindow}.");

    // 3. Current pods
    var currentPods = await GetNamespacePodsAsync(request.Namespace);
    var podCount = Math.Max(currentPods.Count, 1);

    // 4. Derived per-order costs
    double memoryPerOrder = avgMem / historicalOrders;
    double ordersPerPodPerSec =
        historicalOrders / (request.HistoricalWindow.TotalSeconds * podCount);
    double memoryPerPod = avgMem / podCount; // observed per-pod memory

    // 5. Project to the target
    double requiredThroughputPerSec = request.TargetOrders / (request.WithinMinutes * 60.0);

    int podsFromThroughput = (int)Math.Ceiling(
        requiredThroughputPerSec / Math.Max(ordersPerPodPerSec, double.Epsilon));

    // Memory constraint uses P95 (peak), not avg
    double requiredTotalMem = request.TargetOrders * memoryPerOrder;
    double requiredTotalMemWithP95 = p95Mem * (requiredTotalMem / Math.Max(avgMem, 1));
    int podsFromMemory = (int)Math.Ceiling(
        requiredTotalMemWithP95 / Math.Max(memoryPerPod, 1));

    // 6. Apply safety margin on top of max(throughput, memory)
    double safetyFactor = 1.0 + request.SafetyMarginPercent / 100.0;
    int rawPods = Math.Max(podsFromThroughput, podsFromMemory);
    int recommendedPods = (int)Math.Ceiling(rawPods * safetyFactor);

    double recommendedTotalMem = requiredTotalMemWithP95 * safetyFactor;
    double recommendedMemPerPod = recommendedTotalMem / recommendedPods;

    return new CapacityRecommendation
    {
        Namespace = request.Namespace,
        TargetOrders = request.TargetOrders,
        WithinMinutes = request.WithinMinutes,
        HistoricalWindow = request.HistoricalWindow,
        SafetyMarginPercent = request.SafetyMarginPercent,

        CurrentPodCount = podCount,
        HistoricalOrders = historicalOrders,
        HistoricalAvgMemoryBytes = avgMem,
        HistoricalP95MemoryBytes = p95Mem,
        HistoricalMaxMemoryBytes = maxMem,

        MemoryPerOrderBytes = memoryPerOrder,
        HistoricalOrdersPerPodPerSecond = ordersPerPodPerSec,

        RequiredThroughputPerSecond = requiredThroughputPerSec,
        PodsFromThroughput = podsFromThroughput,
        RequiredTotalMemoryBytes = requiredTotalMemWithP95,
        PodsFromMemory = podsFromMemory,

        RecommendedPods = recommendedPods,
        RecommendedMemoryPerPodBytes = recommendedMemPerPod,
        RecommendedTotalMemoryBytes = recommendedTotalMem,
    };
}

// Prometheus side
private async Task<PrometheusRangeResult> GetHistoricalMemoryUsageAsync(
    string namespaceName, DateTime fromUtc, DateTime toUtc)
{
    var start = new DateTimeOffset(fromUtc, TimeSpan.Zero).ToUnixTimeSeconds();
    var end   = new DateTimeOffset(toUtc,   TimeSpan.Zero).ToUnixTimeSeconds();

    var query = $"sum(container_memory_working_set_bytes{{namespace=\"{namespaceName}\",container!=\"POD\",container!=\"\"}})";
    var url = $"{_prometheusUrl}/api/v1/query_range" +
              $"?query={Uri.EscapeDataString(query)}&start={start}&end={end}&step=300";

    var response = await _prometheusClient.GetAsync(url);
    response.EnsureSuccessStatusCode();
    var json = await response.Content.ReadAsStringAsync();
    var result = JsonSerializer.Deserialize<PrometheusResponse>(json,
        new JsonSerializerOptions { PropertyNameCaseInsensitive = true });

    if (result?.Data?.Result == null || !result.Data.Result.Any()) return null;

    var values = new List<MetricValue>();
    foreach (var series in result.Data.Result)
        foreach (var v in series.Values)
            if (double.TryParse(v[1], out var d))
                values.Add(new MetricValue { Timestamp = long.Parse(v[0]), Value = d });

    return new PrometheusRangeResult { Values = values };
}


using k8s;
using k8s.Models;

/// <summary>
/// Returns the pods in a namespace that are actually running/ready.
/// Filters out Succeeded/Failed pods and pods pending deletion so the
/// count reflects real serving capacity.
/// </summary>
private async Task<List<V1Pod>> GetNamespacePodsAsync(
    string namespaceName,
    CancellationToken ct = default)
{
    try
    {
        var podList = await _kubeClient.CoreV1.ListNamespacedPodAsync(
            namespace: namespaceName,
            cancellationToken: ct);

        return podList.Items
            .Where(p => p.Metadata?.DeletionTimestamp == null)
            .Where(p => p.Status?.Phase == "Running")
            .Where(p => IsPodReady(p))
            .ToList();
    }
    catch (k8s.Autorest.HttpOperationException ex) when (ex.Response.StatusCode == System.Net.HttpStatusCode.NotFound)
    {
        throw new InvalidOperationException(
            $"Namespace '{namespaceName}' not found.", ex);
    }
    catch (Exception ex)
    {
        throw new InvalidOperationException(
            $"Failed to list pods in namespace '{namespaceName}': {ex.Message}", ex);
    }
}

/// <summary>
/// True if the pod has a Ready condition with status True.
/// </summary>
private static bool IsPodReady(V1Pod pod)
{
    var conditions = pod.Status?.Conditions;
    if (conditions == null) return false;

    return conditions.Any(c =>
        string.Equals(c.Type, "Ready", StringComparison.OrdinalIgnoreCase) &&
        string.Equals(c.Status, "True", StringComparison.OrdinalIgnoreCase));
}
