using Microsoft.ML.Data;

namespace OrderCapacityPoc;

// ---------- Raw sources ----------
public record OrderCountRecord(DateTime TimestampUtc, int OrderCount);
public record PrometheusSample(DateTime TimestampUtc, double Value);

// One series returned by Prometheus (metric labels + samples)
public record PromSeries(
    Dictionary<string, string> Labels,
    List<PrometheusSample> Samples);

// ---------- Prediction input (features) ----------
public class ModelInput
{
    public float OrdersPerMinute { get; set; }
    public float HourOfDay { get; set; }
    public float DayOfWeek { get; set; }
}

// ---------- Training row: features + labels ----------
public class TrainingRow : ModelInput
{
    public DateTime TimestampUtc { get; set; }
    public float RunningPods { get; set; }
    public float AvgCpuPerPodMillis { get; set; }
    public float AvgMemoryPerPodMiB { get; set; }

    // Labels used by the trainers
    public float Pods { get; set; }
    public float Cpu { get; set; }
    public float Mem { get; set; }
}

// Intermediate schema for feature engineering
public class FeatureRow
{
    public float OrdersPerMinute { get; set; }
    public float OrdersSquared { get; set; }
    public float HourOfDay { get; set; }
    public float DayOfWeek { get; set; }
}

// ---------- Prediction outputs ----------
public class PodPrediction
{
    [ColumnName("Score")] public float PredictedPods { get; set; }
}

public class CpuPrediction
{
    [ColumnName("Score")] public float PredictedCpu { get; set; }
}

public class MemPrediction
{
    [ColumnName("Score")] public float PredictedMem { get; set; }
}

// ---------- Final result ----------
public record PredictionResult(
    DateTime TimestampUtc,
    int ExpectedOrdersPerMinute,
    int RecommendedPods,
    int RecommendedCpuMillisPerPod,
    int RecommendedMemoryMiBPerPod);

public static class SyntheticData
{
    public static List<TrainingRow> Generate(int days = 14, int seed = 42)
    {
        var rng = new Random(seed);
        var start = DateTime.UtcNow.Date.AddDays(-days);
        var rows = new List<TrainingRow>(days * 1440);

        for (int i = 0; i < days * 1440; i++)
        {
            var ts = start.AddMinutes(i);

            // Daily wave: peak ~14:00, trough ~03:00
            double daily = 500 + 400 * Math.Sin((ts.Hour - 8) * Math.PI / 12.0);

            // Weekly: weekends quieter
            double weekly = (ts.DayOfWeek == DayOfWeek.Saturday ||
                             ts.DayOfWeek == DayOfWeek.Sunday) ? 0.55 : 1.0;

            // Random noise
            double noise = (rng.NextDouble() - 0.5) * 60;

            double orders = Math.Max(20, daily * weekly + noise);

            // Ground-truth relationships (what the model should rediscover)
            double pods = Math.Max(2, orders / 250.0 + (rng.NextDouble() - 0.5) * 0.6);
            double cpu  = 120 + 180 * (1 - Math.Exp(-orders / 800.0))
                          + (rng.NextDouble() - 0.5) * 25;
            double mem  = 220 + 180 * (1 - Math.Exp(-orders / 1200.0))
                          + (rng.NextDouble() - 0.5) * 15;

            rows.Add(new TrainingRow
            {
                TimestampUtc = ts,
                OrdersPerMinute = (float)orders,
                HourOfDay = ts.Hour,
                DayOfWeek = (float)ts.DayOfWeek,
                RunningPods = (float)pods,
                AvgCpuPerPodMillis = (float)cpu,
                AvgMemoryPerPodMiB = (float)mem,
                Pods = (float)pods,
                Cpu  = (float)cpu,
                Mem  = (float)mem
            });
        }
        return rows;
    }
}
namespace OrderCapacityPoc;

public class LinearModel
{
    private (double slope, double intercept) _pod;
    private (double slope, double intercept) _cpu;
    private (double slope, double intercept) _mem;

    public void Train(IEnumerable<TrainingRow> rows)
    {
        var list = rows.ToList();
        _pod = Fit(list.Select(r => ((double)r.OrdersPerMinute, (double)r.Pods)));
        _cpu = Fit(list.Select(r => ((double)r.OrdersPerMinute, (double)r.Cpu)));
        _mem = Fit(list.Select(r => ((double)r.OrdersPerMinute, (double)r.Mem)));
    }

    private static (double, double) Fit(IEnumerable<(double X, double Y)> pts)
    {
        var arr = pts.ToList();
        double mx = arr.Average(p => p.X);
        double my = arr.Average(p => p.Y);
        double sxy = arr.Sum(p => (p.X - mx) * (p.Y - my));
        double sxx = arr.Sum(p => (p.X - mx) * (p.X - mx));
        double slope = sxx == 0 ? 0 : sxy / sxx;
        return (slope, my - slope * mx);
    }

    public PredictionResult Predict(int orders, double safety = 1.15)
    {
        double pods = (_pod.slope * orders + _pod.intercept) * safety;
        double cpu  = _cpu.slope * orders + _cpu.intercept;
        double mem  = _mem.slope * orders + _mem.intercept;

        return new PredictionResult(
            DateTime.UtcNow, orders,
            Math.Max(1, (int)Math.Ceiling(pods)),
            Math.Max(50, (int)Math.Round(cpu)),
            Math.Max(64, (int)Math.Round(mem)));
    }
}

public class MlModel
{
    private readonly MLContext _ml = new(seed: 0);
    private ITransformer? _podModel;
    private ITransformer? _cpuModel;
    private ITransformer? _memModel;
    private PredictionEngine<ModelInput, PodPrediction>? _podEngine;
    private PredictionEngine<ModelInput, CpuPrediction>? _cpuEngine;
    private PredictionEngine<ModelInput, MemPrediction>? _memEngine;
    private readonly object _lock = new();

    public (double podR2, double podRmse,
            double cpuR2, double cpuRmse,
            double memR2, double memRmse) Train(
        IEnumerable<TrainingRow> rows, double testFraction = 0.2)
    {
        var data = _ml.Data.LoadFromEnumerable(rows);
        var split = _ml.Data.TrainTestSplit(data, testFraction: testFraction);

        // ---- Shared feature engineering: Orders, Orders²/1000, Hour, DayOfWeek ----
        var featurePipeline = _ml.Transforms.CustomMapping<ModelInput, FeatureRow>(
            (input, output) =>
            {
                output.OrdersPerMinute = input.OrdersPerMinute;
                output.OrdersSquared   = input.OrdersPerMinute * input.OrdersPerMinute / 1000f;
                output.HourOfDay       = input.HourOfDay;
                output.DayOfWeek       = input.DayOfWeek;
            }, contractName: null);

        var featureCols = new[]
        {
            nameof(FeatureRow.OrdersPerMinute),
            nameof(FeatureRow.OrdersSquared),
            nameof(FeatureRow.HourOfDay),
            nameof(FeatureRow.DayOfWeek)
        };

        // ---- POD pipeline ----
        var podPipeline = featurePipeline
            .Append(_ml.Transforms.Concatenate("Features", featureCols))
            .Append(_ml.Regression.Trainers.FastTree(
                labelColumnName: nameof(TrainingRow.Pods),
                numberOfLeaves: 20,
                numberOfTrees: 100,
                minimumExampleCountPerLeaf: 10));

        // ---- CPU pipeline ----
        var cpuPipeline = featurePipeline
            .Append(_ml.Transforms.Concatenate("Features", featureCols))
            .Append(_ml.Regression.Trainers.FastTree(
                labelColumnName: nameof(TrainingRow.Cpu),
                numberOfLeaves: 20,
                numberOfTrees: 100,
                minimumExampleCountPerLeaf: 10));

        // ---- MEM pipeline ----
        var memPipeline = featurePipeline
            .Append(_ml.Transforms.Concatenate("Features", featureCols))
            .Append(_ml.Regression.Trainers.FastTree(
                labelColumnName: nameof(TrainingRow.Mem),
                numberOfLeaves: 20,
                numberOfTrees: 100,
                minimumExampleCountPerLeaf: 10));

        _podModel = podPipeline.Fit(split.TrainSet);
        _cpuModel = cpuPipeline.Fit(split.TrainSet);
        _memModel = memPipeline.Fit(split.TrainSet);

        var podEval = _ml.Regression.Evaluate(
            _podModel.Transform(split.TestSet),
            labelColumnName: nameof(TrainingRow.Pods));
        var cpuEval = _ml.Regression.Evaluate(
            _cpuModel.Transform(split.TestSet),
            labelColumnName: nameof(TrainingRow.Cpu));
        var memEval = _ml.Regression.Evaluate(
            _memModel.Transform(split.TestSet),
            labelColumnName: nameof(TrainingRow.Mem));

        _podEngine = _ml.Model.CreatePredictionEngine<ModelInput, PodPrediction>(_podModel);
        _cpuEngine = _ml.Model.CreatePredictionEngine<ModelInput, CpuPrediction>(_cpuModel);
        _memEngine = _ml.Model.CreatePredictionEngine<ModelInput, MemPrediction>(_memModel);

        return (podEval.RSquared, podEval.RootMeanSquaredError,
                cpuEval.RSquared, cpuEval.RootMeanSquaredError,
                memEval.RSquared, memEval.RootMeanSquaredError);
    }

    public PredictionResult Predict(
        int expectedOrdersPerMinute,
        double safetyFactor = 1.15,
        int hourOfDay = 12,
        int dayOfWeek = 3)
    {
        if (_podEngine is null || _cpuEngine is null || _memEngine is null)
            throw new InvalidOperationException("Model not trained. Call Train() first.");

        var input = new ModelInput
        {
            OrdersPerMinute = expectedOrdersPerMinute,
            HourOfDay = hourOfDay,
            DayOfWeek = dayOfWeek
        };

        float pods, cpu, mem;
        lock (_lock) // PredictionEngine is not thread-safe
        {
            pods = _podEngine.Predict(input).PredictedPods;
            cpu  = _cpuEngine.Predict(input).PredictedCpu;
            mem  = _memEngine.Predict(input).PredictedMem;
        }

        return new PredictionResult(
            DateTime.UtcNow,
            expectedOrdersPerMinute,
            Math.Max(1,  (int)Math.Ceiling(pods * safetyFactor)),
            Math.Max(50, (int)Math.Round(cpu)),
            Math.Max(64, (int)Math.Round(mem * safetyFactor)));
    }
}

using System.Net.Http.Json;
using System.Text.Json.Serialization;

namespace OrderCapacityPoc;

public class PrometheusService
{
    private readonly HttpClient _http;

    public PrometheusService(HttpClient http, string baseUrl)
    {
        _http = http;
        if (_http.BaseAddress is null) _http.BaseAddress = new Uri(baseUrl);
    }

    // -------------------------------------------------------------
    // Low-level: returns ALL series Prometheus returned.
    // Use this when you want to inspect cardinality or pick a subset.
    // -------------------------------------------------------------
    public async Task<List<PromSeries>> QueryRangeAllAsync(
        string promQl, DateTime startUtc, DateTime endUtc,
        int stepSeconds = 60, CancellationToken ct = default)
    {
        var url = $"/api/v1/query_range?query={Uri.EscapeDataString(promQl)}" +
                  $"&start={new DateTimeOffset(startUtc).ToUnixTimeSeconds()}" +
                  $"&end={new DateTimeOffset(endUtc).ToUnixTimeSeconds()}" +
                  $"&step={stepSeconds}";

        var resp = await _http.GetFromJsonAsync<PromResponse>(url, ct);
        var list = new List<PromSeries>();

        if (resp?.Data?.Result is null || resp.Data.Result.Count == 0)
            return list;

        foreach (var r in resp.Data.Result)
        {
            var samples = new List<PrometheusSample>(r.Values.Count);
            foreach (var pair in r.Values)
            {
                var ts = DateTimeOffset.FromUnixTimeSeconds((long)pair[0]).UtcDateTime;
                if (double.TryParse(pair[1], out var val) && !double.IsNaN(val))
                    samples.Add(new PrometheusSample(ts, val));
            }
            list.Add(new PromSeries(r.Metric ?? new(), samples));
        }
        return list;
    }

    // -------------------------------------------------------------
    // Convenience: aggregates across all returned series per timestamp.
    // agg: "avg" | "sum" | "max" | "min" | "count"
    // -------------------------------------------------------------
    public async Task<List<PrometheusSample>> QueryRangeAggregatedAsync(
        string promQl, DateTime startUtc, DateTime endUtc,
        string agg = "avg", int stepSeconds = 60, CancellationToken ct = default)
    {
        var all = await QueryRangeAllAsync(promQl, startUtc, endUtc, stepSeconds, ct);
        return AggregateByTimestamp(all, agg);
    }

    private static List<PrometheusSample> AggregateByTimestamp(
        List<PromSeries> series, string agg)
    {
        var buckets = new Dictionary<DateTime, List<double>>();
        foreach (var s in series)
            foreach (var sample in s.Samples)
            {
                if (!buckets.TryGetValue(sample.TimestampUtc, out var list))
                    buckets[sample.TimestampUtc] = list = new List<double>();
                list.Add(sample.Value);
            }

        var result = new List<PrometheusSample>(buckets.Count);
        foreach (var kv in buckets.OrderBy(kv => kv.Key))
        {
            var vals = kv.Value;
            double v = agg switch
            {
                "sum"   => vals.Sum(),
                "max"   => vals.Max(),
                "min"   => vals.Min(),
                "count" => vals.Count,
                _       => vals.Average()
            };
            result.Add(new PrometheusSample(kv.Key, v));
        }
        return result;
    }

    // -------------------------------------------------------------
    // Public metric helpers — all use aggregated queries
    // -------------------------------------------------------------

    public Task<List<PrometheusSample>> GetRunningPodsAsync(
        string ns, string deployment, DateTime from, DateTime to,
        CancellationToken ct = default)
        => QueryRangeAggregatedAsync(
            $"sum(kube_deployment_status_replicas{{namespace=\"{ns}\",deployment=\"{deployment}\"}})",
            from, to, "sum", 60, ct);

    public Task<List<PrometheusSample>> GetAvgCpuPerPodMillisAsync(
        string ns, string podRegex, DateTime from, DateTime to,
        CancellationToken ct = default)
        => QueryRangeAggregatedAsync(
            "avg(" +
            "  sum by (pod) (" +
            $"    rate(container_cpu_usage_seconds_total{{namespace=\"{ns}\",pod=~\"{podRegex}\",container!=\"\",container!=\"POD\"}}[1m])" +
            "  ) * 1000" +
            ")",
            from, to, "avg", 60, ct);

    public Task<List<PrometheusSample>> GetAvgMemoryPerPodMiBAsync(
        string ns, string podRegex, DateTime from, DateTime to,
        CancellationToken ct = default)
        => QueryRangeAggregatedAsync(
            "avg(" +
            "  sum by (pod) (" +
            $"    container_memory_working_set_bytes{{namespace=\"{ns}\",pod=~\"{podRegex}\",container!=\"\",container!=\"POD\"}}" +
            "  ) / 1024 / 1024" +
            ")",
            from, to, "avg", 60, ct);

    // -------------------------------------------------------------
    // Diagnostic helper — prints each series with its labels
    // -------------------------------------------------------------
    public async Task DumpSeriesAsync(
        string promQl, DateTime from, DateTime to,
        CancellationToken ct = default)
    {
        var all = await QueryRangeAllAsync(promQl, from, to, 60, ct);
        Console.WriteLine($"Query returned {all.Count} series:");
        foreach (var s in all)
        {
            var labelStr = string.Join(", ", s.Labels.Select(kv => $"{kv.Key}={kv.Value}"));
            var first = s.Samples.FirstOrDefault();
            Console.WriteLine($"  [{s.Samples.Count,5} pts] {labelStr}  first={first.Value:F2}@{first.TimestampUtc:HH:mm}");
        }
    }

    // -------------------------------------------------------------
    // DTOs for JSON deserialization
    // -------------------------------------------------------------
    private class PromResponse
    {
        [JsonPropertyName("data")] public PromData? Data { get; set; }
    }
    private class PromData
    {
        [JsonPropertyName("result")] public List<PromResult> Result { get; set; } = new();
    }
    private class PromResult
    {
        [JsonPropertyName("metric")] public Dictionary<string, string>? Metric { get; set; }
        [JsonPropertyName("values")] public List<List<object>> Values { get; set; } = new();
    }
}

public static class DataAligner
{
    /// <summary>
    /// Joins order counts, pod counts, CPU, and memory on minute buckets.
    /// <paramref name="podsLag"/> shifts pod lookups forward, compensating
    /// for HPA reaction time (pods scale ~1–2 min AFTER orders arrive).
    /// </summary>
    public static List<TrainingRow> Align(
        List<OrderCountRecord> orders,
        List<PrometheusSample> pods,
        List<PrometheusSample> cpu,
        List<PrometheusSample> mem,
        TimeSpan tolerance,
        TimeSpan podsLag = default)
    {
        var podLookup = pods.ToDictionary(p => RoundToMinute(p.TimestampUtc), p => p.Value);
        var cpuLookup = cpu.ToDictionary(c => RoundToMinute(c.TimestampUtc), c => c.Value);
        var memLookup = mem.ToDictionary(m => RoundToMinute(m.TimestampUtc), m => m.Value);

        var rows = new List<TrainingRow>();
        foreach (var o in orders)
        {
            var key = RoundToMinute(o.TimestampUtc);
            var podKey = key.Add(podsLag);

            var podVal = FindNearest(podLookup, podKey, tolerance);
            var cpuVal = FindNearest(cpuLookup, key, tolerance);
            var memVal = FindNearest(memLookup, key, tolerance);

            if (podVal is null || cpuVal is null || memVal is null) continue;

            rows.Add(new TrainingRow
            {
                TimestampUtc = key,
                OrdersPerMinute = o.OrderCount,
                HourOfDay = key.Hour,
                DayOfWeek = (float)key.DayOfWeek,
                RunningPods = (float)podVal.Value,
                AvgCpuPerPodMillis = (float)cpuVal.Value,
                AvgMemoryPerPodMiB = (float)memVal.Value,
                Pods = (float)podVal.Value,
                Cpu  = (float)cpuVal.Value,
                Mem  = (float)memVal.Value
            });
        }
        return rows;
    }

    private static DateTime RoundToMinute(DateTime dt) =>
        new(dt.Year, dt.Month, dt.Day, dt.Hour, dt.Minute, 0, DateTimeKind.Utc);

    private static double? FindNearest(
        Dictionary<DateTime, double> lookup, DateTime key, TimeSpan tol)
    {
        if (lookup.TryGetValue(key, out var exact)) return exact;
        for (int i = 1; i <= tol.TotalMinutes; i++)
        {
            if (lookup.TryGetValue(key.AddMinutes(-i), out var prev)) return prev;
            if (lookup.TryGetValue(key.AddMinutes(i), out var next)) return next;
        }
        return null;
    }
}

    var orderSvc = new OrderDataService(sqlConn);
    var promSvc  = new PrometheusService(new HttpClient(), promUrl);

    var to   = DateTime.UtcNow;

    var pods   = await promSvc.GetRunningPodsAsync(ns, deployment, from, to);
    var cpu    = await promSvc.GetAvgCpuPerPodMillisAsync(ns, podRegex, from, to);
    var mem    = await promSvc.GetAvgMemoryPerPodMiBAsync(ns, podRegex, from, to);

    Console.WriteLine($"orders={orders.Count}, pods={pods.Count}, cpu={cpu.Count}, mem={mem.Count}");

    // ---- Lag search: find the shift that maximizes r(orders, pods) ----
    Console.WriteLine("\n--- Lag search (orders → pods) ---");
    TimeSpan bestLag = TimeSpan.Zero;
    double bestR = -1;
    foreach (var lag in new[] { 0, 1, 2, 3, 5 })
    {
        var cand = DataAligner.Align(orders, pods, cpu, mem,
            TimeSpan.FromMinutes(3), TimeSpan.FromMinutes(lag));
        if (cand.Count < 100)
        {
            Console.WriteLine($"lag={lag}m  rows={cand.Count} (too few)");
            continue;
        }
        double r = Pearson(
            cand.Select(x => (double)x.OrdersPerMinute),
            cand.Select(x => (double)x.Pods));
        Console.WriteLine($"lag={lag}m  rows={cand.Count,6}  r(orders,pods)={r,7:F3}");
        if (r > bestR) { bestR = r; bestLag = TimeSpan.FromMinutes(lag); }
    }
    Console.WriteLine($"Best lag = {bestLag.TotalMinutes} min (r={bestR:F3})");

    rows = DataAligner.Align(orders, pods, cpu, mem,
        TimeSpan.FromMinutes(3), bestLag);
    Console.WriteLine($"Aligned rows: {rows.Count}");
}

static void PrintStats(string name, IEnumerable<float> values)
{
    var v = values.ToList();
    double m = v.Average();
    double sd = Math.Sqrt(v.Sum(x => (x - m) * (x - m)) / v.Count);
    Console.WriteLine($"{name,-7} min={v.Min(),8:F2} max={v.Max(),8:F2} mean={m,8:F2} std={sd,7:F2}");
}

static double Pearson(IEnumerable<double> xs, IEnumerable<double> ys)
{
    var a = xs.ToList();
    var b = ys.ToList();
    if (a.Count == 0 || a.Count != b.Count) return 0;
    double mx = a.Average(), my = b.Average();
    double sxy = a.Zip(b, (x, y) => (x - mx) * (y - my)).Sum();
    double sxx = a.Sum(x => (x - mx) * (x - mx));
    double syy = b.Sum(y => (y - my) * (y - my));
    if (sxx == 0 || syy == 0) return 0;
    return sxy / Math.Sqrt(sxx * syy);
}
