// Models/OrderCountRecord.cs
public record OrderCountRecord(DateTime TimestampUtc, int OrderCount);

// Models/PrometheusSample.cs
public record PrometheusSample(DateTime TimestampUtc, double Value);

// Models/TrainingRow.cs
public class TrainingRow
{
    public DateTime TimestampUtc { get; set; }
    public float OrdersPerMinute { get; set; }
    public float RunningPods { get; set; }
    public float AvgCpuPerPodMillis { get; set; } // e.g. 250m = 250
    public float TotalCpuMillis => RunningPods * AvgCpuPerPodMillis;

    // ML.NET labels
    public float Label_Pods { get; set; }
    public float Label_Cpu { get; set; }
}

// Models/PredictionResult.cs
public record PredictionResult(
    DateTime TimestampUtc,
    int ExpectedOrdersPerMinute,
    int RecommendedPods,
    int RecommendedCpuMillisPerPod);


using System.Net.Http.Json;
using System.Text.Json.Serialization;

public class PrometheusService
{
    private readonly HttpClient _http;
    public PrometheusService(HttpClient http, string baseUrl)
    {
        _http = http;
        _http.BaseAddress = new Uri(baseUrl);
    }

    public async Task<List<PrometheusSample>> QueryRangeAsync(
        string promQl, DateTime startUtc, DateTime endUtc,
        int stepSeconds = 60, CancellationToken ct = default)
    {
        var url = $"/api/v1/query_range?query={Uri.EscapeDataString(promQl)}" +
                  $"&start={new DateTimeOffset(startUtc).ToUnixTimeSeconds()}" +
                  $"&end={new DateTimeOffset(endUtc).ToUnixTimeSeconds()}" +
                  $"&step={stepSeconds}";

        var resp = await _http.GetFromJsonAsync<PromResponse>(url, ct);
        var samples = new List<PrometheusSample>();
        if (resp?.Data?.Result is null || resp.Data.Result.Count == 0)
            return samples;

        // Take the first time-series returned (adjust if you need to aggregate)
        foreach (var pair in resp.Data.Result[0].Values)
        {
            var ts = DateTimeOffset.FromUnixTimeSeconds((long)pair[0]).UtcDateTime;
            var val = double.Parse(pair[1]);
            samples.Add(new PrometheusSample(ts, val));
        }
        return samples;
    }

    // Convenience wrappers for the three metrics we need
    public Task<List<PrometheusSample>> GetRunningPodsAsync(
        string namespaceName, string deployment, DateTime from, DateTime to, CancellationToken ct = default)
    {
        // Requires kube-state-metrics
        var q = $"kube_deployment_status_replicas{{namespace=\"{namespaceName}\",deployment=\"{deployment}\"}}";
        return QueryRangeAsync(q, from, to, 60, ct);
    }

    public Task<List<PrometheusSample>> GetAvgCpuPerPodMillisAsync(
        string namespaceName, string podRegex, DateTime from, DateTime to, CancellationToken ct = default)
    {
        // Rate over 1m, per pod, then average across pods, in millicores
        var q = $"avg(rate(container_cpu_usage_seconds_total{{namespace=\"{namespaceName}\",pod=~\"{podRegex}\",container!=\"\"}}[1m])) * 1000";
        return QueryRangeAsync(q, from, to, 60, ct);
    }

    // ---- Response DTOs ----
    private class PromResponse { [JsonPropertyName("data")] public PromData? Data { get; set; } }
    private class PromData { [JsonPropertyName("result")] public List<PromResult> Result { get; set; } = new(); }
    private class PromResult { [JsonPropertyName("values")] public List<List<object>> Values { get; set; } = new(); }
}




public static class DataAligner
{
    /// <summary>
    /// Joins order counts, pod counts, and CPU samples on minute buckets.
    /// Uses nearest-sample matching within a tolerance window.
    /// </summary>
    public static List<TrainingRow> Align(
        List<OrderCountRecord> orders,
        List<PrometheusSample> pods,
        List<PrometheusSample> cpu,
        TimeSpan tolerance)
    {
        var podLookup = pods.ToDictionary(p => RoundToMinute(p.TimestampUtc), p => p.Value);
        var cpuLookup = cpu.ToDictionary(c => RoundToMinute(c.TimestampUtc), c => c.Value);

        var rows = new List<TrainingRow>();
        foreach (var o in orders)
        {
            var key = RoundToMinute(o.TimestampUtc);

            // Try exact minute, else search nearby within tolerance
            double? podVal = FindNearest(podLookup, key, tolerance);
            double? cpuVal = FindNearest(cpuLookup, key, tolerance);

            if (podVal is null || cpuVal is null) continue;

            rows.Add(new TrainingRow
            {
                TimestampUtc = key,
                OrdersPerMinute = o.OrderCount,
                RunningPods = (float)podVal.Value,
                AvgCpuPerPodMillis = (float)cpuVal.Value,
                Label_Pods = (float)podVal.Value,
                Label_Cpu = (float)cpuVal.Value
            });
        }
        return rows;
    }

    private static DateTime RoundToMinute(DateTime dt) =>
        new DateTime(dt.Year, dt.Month, dt.Day, dt.Hour, dt.Minute, 0, DateTimeKind.Utc);

    private static double? FindNearest(
        Dictionary<DateTime, double> lookup, DateTime key, TimeSpan tolerance)
    {
        if (lookup.TryGetValue(key, out var exact)) return exact;
        for (int offset = 1; offset <= tolerance.TotalMinutes; offset++)
        {
            if (lookup.TryGetValue(key.AddMinutes(-offset), out var prev)) return prev;
            if (lookup.TryGetValue(key.AddMinutes(offset), out var next)) return next;
        }
        return null;
    }
}

public static class CorrelationAnalyzer
{
    public record CorrResult(double PearsonR, double Slope, double Intercept);

    public static CorrResult Analyze(IEnumerable<(double X, double Y)> points)
    {
        var list = points.ToList();
        int n = list.Count;
        if (n < 2) return new CorrResult(0, 0, 0);

        double meanX = list.Average(p => p.X);
        double meanY = list.Average(p => p.Y);
        double sxy = list.Sum(p => (p.X - meanX) * (p.Y - meanY));
        double sxx = list.Sum(p => Math.Pow(p.X - meanX, 2));
        double syy = list.Sum(p => Math.Pow(p.Y - meanY, 2));

        double r = sxy / Math.Sqrt(sxx * syy);
        double slope = sxy / sxx;
        double intercept = meanY - slope * meanX;
        return new CorrResult(r, slope, intercept);
    }
}

public class LinearPredictionModel
{
    private readonly CorrelationAnalyzer.CorrResult _podModel;
    private readonly CorrelationAnalyzer.CorrResult _cpuModel;

    public LinearPredictionModel(IEnumerable<TrainingRow> rows)
    {
        _podModel = CorrelationAnalyzer.Analyze(rows.Select(r => ((double)r.OrdersPerMinute, (double)r.RunningPods)));
        _cpuModel = CorrelationAnalyzer.Analyze(rows.Select(r => ((double)r.OrdersPerMinute, (double)r.AvgCpuPerPodMillis)));
    }

    public PredictionResult Predict(int expectedOrdersPerMinute, double safetyFactor = 1.15)
    {
        double pods = (_podModel.Slope * expectedOrdersPerMinute + _podModel.Intercept) * safetyFactor;
        double cpu = _cpuModel.Slope * expectedOrdersPerMinute + _cpuModel.Intercept;

        return new PredictionResult(
            DateTime.UtcNow,
            expectedOrdersPerMinute,
            Math.Max(1, (int)Math.Ceiling(pods)),
            Math.Max(50, (int)Math.Round(cpu)));
    }
}

using Microsoft.ML;
using Microsoft.ML.Data;

public class MlPredictionModel
{
    private readonly MLContext _ml = new(seed: 0);
    private ITransformer? _podModel;
    private ITransformer? _cpuModel;

    public class Input
    {
        [LoadColumn(0)] public float OrdersPerMinute { get; set; }
    }
    public class PodOutput
    {
        [ColumnName("Score")] public float PredictedPods { get; set; }
    }
    public class CpuOutput
    {
        [ColumnName("Score")] public float PredictedCpu { get; set; }
    }

    public void Train(IEnumerable<TrainingRow> rows)
    {
        var data = _ml.Data.LoadFromEnumerable(rows);
        var split = _ml.Data.TrainTestSplit(data, testFraction: 0.2);

        // Pod model
        var podPipeline = _ml.Transforms.CopyColumns("Label", nameof(TrainingRow.Label_Pods))
            .Append(_ml.Transforms.Concatenate("Features", nameof(TrainingRow.OrdersPerMinute)))
            .Append(_ml.Regression.Trainers.Sdca(labelColumnName: "Label", maximumNumberOfIterations: 100));
        _podModel = podPipeline.Fit(split.TrainSet);

        // CPU model
        var cpuPipeline = _ml.Transforms.CopyColumns("Label", nameof(TrainingRow.Label_Cpu))
            .Append(_ml.Transforms.Concatenate("Features", nameof(TrainingRow.OrdersPerMinute)))
            .Append(_ml.Regression.Trainers.Sdca(labelColumnName: "Label", maximumNumberOfIterations: 100));
        _cpuModel = cpuPipeline.Fit(split.TrainSet);

        // Quick eval
        var podEval = _ml.Regression.Evaluate(_podModel.Transform(split.TestSet), "Label");
        var cpuEval = _ml.Regression.Evaluate(_cpuModel.Transform(split.TestSet), "Label");
        Console.WriteLine($"[ML] Pod R²={podEval.RSquared:F3} RMSE={podEval.RootMeanSquaredError:F2}");
        Console.WriteLine($"[ML] CPU R²={cpuEval.RSquared:F3} RMSE={cpuEval.RootMeanSquaredError:F2}");
    }

    public PredictionResult Predict(int expectedOrdersPerMinute, double safetyFactor = 1.15)
    {
        if (_podModel is null || _cpuModel is null)
            throw new InvalidOperationException("Model not trained.");

        var input = new Input { OrdersPerMinute = expectedOrdersPerMinute };
        var podPred = _ml.Model.CreatePredictionEngine<Input, PodOutput>(_podModel)
            .Predict(input).PredictedPods * (float)safetyFactor;
        var cpuPred = _ml.Model.CreatePredictionEngine<Input, CpuOutput>(_cpuModel)
            .Predict(input).PredictedCpu;

        return new PredictionResult(
            DateTime.UtcNow,
            expectedOrdersPerMinute,
            Math.Max(1, (int)Math.Ceiling(podPred)),
            Math.Max(50, (int)Math.Round(cpuPred)));
    }
}

using Prometheus;

public class HpaMetricExporter
{
    private readonly Gauge _predictedPods = Metrics.CreateGauge(
        "predicted_pods_required",
        "Recommended pod count based on predicted order volume.");

    private readonly Gauge _predictedCpu = Metrics.CreateGauge(
        "predicted_cpu_millis_required",
        "Recommended CPU per pod (millicores) based on predicted order volume.");

    public void Publish(PredictionResult result)
    {
        _predictedPods.Set(result.RecommendedPods);
        _predictedCpu.Set(result.RecommendedCpuMillisPerPod);
    }

    public static void StartHttpServer(int port = 9095)
    {
        var server = new KestrelMetricServer(port: port);
        server.Start();
    }
}


using Microsoft.Extensions.DependencyInjection;

var services = new ServiceCollection();
services.AddHttpClient<PrometheusService>(c =>
    c.BaseAddress = new Uri("http://prometheus.monitoring.svc:9090"));

var sp = services.BuildServiceProvider();

// ---- CONFIG ----
const string sqlConn = "Server=...;Database=Orders;Trusted_Connection=True;";
const string ns = "production";
const string deployment = "order-service";
const string podRegex = "order-service-.*";

var orderSvc = new OrderDataService(sqlConn);
var promSvc = sp.GetRequiredService<PrometheusService>();

var to = DateTime.UtcNow;
var from = to.AddDays(-14); // 2 weeks of history

Console.WriteLine($"Fetching orders from {from:u} to {to:u}");
var orders = await orderSvc.GetOrderCountsAsync(from, to);

Console.WriteLine("Fetching Prometheus metrics...");
var pods = await promSvc.GetRunningPodsAsync(ns, deployment, from, to);
var cpu = await promSvc.GetAvgCpuPerPodMillisAsync(ns, podRegex, from, to);

Console.WriteLine($"Orders: {orders.Count}, Pods: {pods.Count}, CPU: {cpu.Count}");

// ---- ALIGN ----
var rows = DataAligner.Align(orders, pods, cpu, TimeSpan.FromMinutes(3));
Console.WriteLine($"Aligned rows: {rows.Count}");
if (rows.Count < 100) { Console.WriteLine("Not enough data."); return; }

// ---- CORRELATION ----
var podCorr = CorrelationAnalyzer.Analyze(rows.Select(r => ((double)r.OrdersPerMinute, (double)r.RunningPods)));
var cpuCorr = CorrelationAnalyzer.Analyze(rows.Select(r => ((double)r.OrdersPerMinute, (double)r.AvgCpuPerPodMillis)));
Console.WriteLine($"[Corr] Orders↔Pods r={podCorr.PearsonR:F3}");
Console.WriteLine($"[Corr] Orders↔CPU  r={cpuCorr.PearsonR:F3}");

// ---- TRAIN ----
var model = new MlPredictionModel();
model.Train(rows);

// ---- PREDICT FOR FUTURE WORKLOAD ----
// e.g., business forecast says 1200 orders/min expected
foreach (var expected in new[] { 200, 500, 1000, 2000 })
{
    var pred = model.Predict(expected);
    Console.WriteLine($"Expected {expected} orders/min → {pred.RecommendedPods} pods @ {pred.RecommendedCpuMillisPerPod}m CPU");
}

// ---- EXPOSE FOR HPA ----
HpaMetricExporter.StartHttpServer(9095);
var exporter = new HpaMetricExporter();

// In real life, run this loop in a BackgroundService and read the
// expected order volume from your forecasting system (or read the
// latest minute's orders from SQL as a proxy).
while (true)
{
    var latestOrders = (await orderSvc.GetOrderCountsAsync(
        DateTime.UtcNow.AddMinutes(-5), DateTime.UtcNow))
        .OrderByDescending(o => o.TimestampUtc)
        .FirstOrDefault()?.OrderCount ?? 100;

    var result = model.Predict(latestOrders);
    exporter.Publish(result);
    Console.WriteLine($"[{DateTime.UtcNow:HH:mm:ss}] Orders={latestOrders} → Pods={result.RecommendedPods}");

    await Task.Delay(TimeSpan.FromSeconds(30));
}
