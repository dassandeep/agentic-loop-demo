namespace OrderCapacityPoc;

// ---- Raw data from SQL ----
public record OrderCountRecord(DateTime TimestampUtc, int OrderCount);

// ---- Raw data from Prometheus ----
public record PrometheusSample(DateTime TimestampUtc, double Value);

// ---- Prediction input (features only) ----
public class ModelInput
{
    public float OrdersPerMinute { get; set; }
}

// ---- Training row: inherits input + adds labels ----
public class TrainingRow : ModelInput
{
    public DateTime TimestampUtc { get; set; }
    public float RunningPods { get; set; }
    public float AvgCpuPerPodMillis { get; set; }

    // Labels used by the trainers
    public float Pods { get; set; }
    public float Cpu { get; set; }
}

// ---- Prediction outputs ----
public class PodPrediction
{
    [ColumnName("Score")] public float PredictedPods { get; set; }
}

public class CpuPrediction
{
    [ColumnName("Score")] public float PredictedCpu { get; set; }
}

// ---- Final result ----
public record PredictionResult(
    DateTime TimestampUtc,
    int ExpectedOrdersPerMinute,
    int RecommendedPods,
    int RecommendedCpuMillisPerPod);



namespace OrderCapacityPoc;

public static class SyntheticData
{
    /// <summary>
    /// Generates 14 days of minute-by-minute orders with:
    ///  - a daily sine wave (peak midday)
    ///  - a weekly wave (busy on weekdays)
    ///  - random noise
    /// Pods and CPU are derived from orders with a known-ish relationship
    /// plus noise, so the model has something to learn.
    /// </summary>
    public static List<TrainingRow> Generate(int days = 14, int seed = 42)
    {
        var rng = new Random(seed);
        var start = DateTime.UtcNow.Date.AddDays(-days);
        var rows = new List<TrainingRow>(days * 1440);

        for (int i = 0; i < days * 1440; i++)
        {
            var ts = start.AddMinutes(i);

            // Daily pattern: peak around 14:00, low around 03:00
            double daily = 500 + 400 * Math.Sin((ts.Hour - 8) * Math.PI / 12.0);

            // Weekly pattern: weekends quieter
            double weekly = (ts.DayOfWeek == DayOfWeek.Saturday || ts.DayOfWeek == DayOfWeek.Sunday)
                ? 0.55 : 1.0;

            // Noise
            double noise = (rng.NextDouble() - 0.5) * 60;

            double orders = Math.Max(20, daily * weekly + noise);

            // Ground-truth relationships (what we want the model to rediscover)
            // pods ≈ orders / 250  (with a floor of 2), plus a little noise
            double pods = Math.Max(2, orders / 250.0 + (rng.NextDouble() - 0.5) * 0.6);

            // cpu per pod in millicores: rises with load but with diminishing returns
            double cpu = 120 + 180 * (1 - Math.Exp(-orders / 800.0))
                         + (rng.NextDouble() - 0.5) * 25;

            rows.Add(new TrainingRow
            {
                TimestampUtc = ts,
                OrdersPerMinute = (float)orders,
                RunningPods = (float)pods,
                AvgCpuPerPodMillis = (float)cpu,
                Pods = (float)pods,
                Cpu = (float)cpu
            });
        }
        return rows;
    }
}

namespace OrderCapacityPoc;

public class LinearModel
{
    private (double slope, double intercept) _pod;
    private (double slope, double intercept) _cpu;

    public void Train(IEnumerable<TrainingRow> rows)
    {
        var list = rows.ToList();
        _pod = Fit(list.Select(r => ((double)r.OrdersPerMinute, (double)r.Pods)));
        _cpu = Fit(list.Select(r => ((double)r.OrdersPerMinute, (double)r.Cpu)));
    }

    private static (double, double) Fit(IEnumerable<(double X, double Y)> pts)
    {
        var arr = pts.ToList();
        double mx = arr.Average(p => p.X);
        double my = arr.Average(p => p.Y);
        double sxy = arr.Sum(p => (p.X - mx) * (p.Y - my));
        double sxx = arr.Sum(p => (p.X - mx) * (p.X - mx));
        double slope = sxx == 0 ? 0 : sxy / sxx;
        return (slope, my - slope * mx);
    }

    public PredictionResult Predict(int orders, double safety = 1.15)
    {
        double pods = (_pod.slope * orders + _pod.intercept) * safety;
        double cpu = _cpu.slope * orders + _cpu.intercept;
        return new PredictionResult(
            DateTime.UtcNow, orders,
            Math.Max(1, (int)Math.Ceiling(pods)),
            Math.Max(50, (int)Math.Round(cpu)));
    }
}
using Microsoft.ML;
using Microsoft.ML.Data;

namespace OrderCapacityPoc;

public class MlModel
{
    private readonly MLContext _ml = new(seed: 0);
    private ITransformer? _podModel;
    private ITransformer? _cpuModel;
    private PredictionEngine<ModelInput, PodPrediction>? _podEngine;
    private PredictionEngine<ModelInput, CpuPrediction>? _cpuEngine;
    private readonly object _lock = new();

    public (double podR2, double podRmse, double cpuR2, double cpuRmse) Train(
        IEnumerable<TrainingRow> rows, double testFraction = 0.2)
    {
        var data = _ml.Data.LoadFromEnumerable(rows);
        var split = _ml.Data.TrainTestSplit(data, testFraction: testFraction);

        // ---- POD pipeline ----
        var podPipeline = _ml.Transforms
            .Concatenate("Features", nameof(ModelInput.OrdersPerMinute))
            .Append(_ml.Regression.Trainers.Sdca(
                labelColumnName: nameof(TrainingRow.Pods),
                maximumNumberOfIterations: 200));

        // ---- CPU pipeline ----
        var cpuPipeline = _ml.Transforms
            .Concatenate("Features", nameof(ModelInput.OrdersPerMinute))
            .Append(_ml.Regression.Trainers.Sdca(
                labelColumnName: nameof(TrainingRow.Cpu),
                maximumNumberOfIterations: 200));

        _podModel = podPipeline.Fit(split.TrainSet);
        _cpuModel = cpuPipeline.Fit(split.TrainSet);

        var podEval = _ml.Regression.Evaluate(
            _podModel.Transform(split.TestSet), labelColumnName: nameof(TrainingRow.Pods));
        var cpuEval = _ml.Regression.Evaluate(
            _cpuModel.Transform(split.TestSet), labelColumnName: nameof(TrainingRow.Cpu));

        // Recreate engines after training
        _podEngine = _ml.Model.CreatePredictionEngine<ModelInput, PodPrediction>(_podModel);
        _cpuEngine = _ml.Model.CreatePredictionEngine<ModelInput, CpuPrediction>(_cpuModel);

        return (podEval.RSquared, podEval.RootMeanSquaredError,
                cpuEval.RSquared, cpuEval.RootMeanSquaredError);
    }

    public PredictionResult Predict(int expectedOrdersPerMinute, double safetyFactor = 1.15)
    {
        if (_podEngine is null || _cpuEngine is null)
            throw new InvalidOperationException("Model not trained.");

        var input = new ModelInput { OrdersPerMinute = expectedOrdersPerMinute };

        float pods, cpu;
        lock (_lock) // PredictionEngine is not thread-safe
        {
            pods = _podEngine.Predict(input).PredictedPods;
            cpu = _cpuEngine.Predict(input).PredictedCpu;
        }

        return new PredictionResult(
            DateTime.UtcNow,
            expectedOrdersPerMinute,
            Math.Max(1, (int)Math.Ceiling(pods * safetyFactor)),
            Math.Max(50, (int)Math.Round(cpu)));
    }
}


using OrderCapacityPoc;

Console.WriteLine("=== Order Capacity POC ===\n");

// 1. Load data (synthetic for the demo; swap for SQL + Prometheus later)
var rows = SyntheticData.Generate(days: 14);
Console.WriteLine($"Generated {rows.Count} minute rows over 14 days.");

// 2. Sanity: split train/test by time (walk-forward style)
var ordered = rows.OrderBy(r => r.TimestampUtc).ToList();
int splitIdx = (int)(ordered.Count * 0.8);
var train = ordered.Take(splitIdx).ToList();
var test = ordered.Skip(splitIdx).ToList();

// 3. Train both models
Console.WriteLine("\n--- Linear baseline ---");
var linear = new LinearModel();
linear.Train(train);

Console.WriteLine("\n--- ML.NET model ---");
var ml = new MlModel();
var (podR2, podRmse, cpuR2, cpuRmse) = ml.Train(ordered); // ML.NET does its own split
Console.WriteLine($"Pod  R²={podR2:F3}  RMSE={podRmse:F2}");
Console.WriteLine($"CPU  R²={cpuR2:F3}  RMSE={cpuRmse:F2}");

// 4. Predict for a range of expected order volumes
Console.WriteLine("\n--- Predictions for expected order volumes ---");
Console.WriteLine($"{"Orders/min",12} | {"Linear pods",12} | {"ML pods",10} | {"ML cpu(m)",10}");
Console.WriteLine(new string('-', 60));

foreach (var orders in new[] { 100, 300, 600, 1000, 1500, 2000 })
{
    var lin = linear.Predict(orders);
    var m   = ml.Predict(orders);
    Console.WriteLine($"{orders,12} | {lin.RecommendedPods,12} | {m.RecommendedPods,10} | {m.RecommendedCpuMillisPerPod,10}");
}

// 5. Backtest on the held-out tail: compare prediction vs actual
Console.WriteLine("\n--- Backtest on last 20% of data ---");
double podAbsErr = 0, cpuAbsErr = 0;
int n = 0;
foreach (var row in test)
{
    var p = ml.Predict((int)row.OrdersPerMinute);
    podAbsErr += Math.Abs(p.RecommendedPods - row.Pods);
    cpuAbsErr += Math.Abs(p.RecommendedCpuMillisPerPod - row.Cpu);
    n++;
}
Console.WriteLine($"Mean abs error: pods={podAbsErr / n:F2}, cpu={cpuAbsErr / n:F1} m");

// 6. Show a few concrete predictions
Console.WriteLine("\n--- Sample predictions (first 10 rows of test set) ---");
Console.WriteLine($"{"Time (UTC)",20} | {"Orders",7} | {"ActualPods",10} | {"PredPods",9} | {"ActualCPU",10} | {"PredCPU",8}");
Console.WriteLine(new string('-', 90));
foreach (var row in test.Take(10))
{
    var p = ml.Predict((int)row.OrdersPerMinute);
    Console.WriteLine($"{row.TimestampUtc:yyyy-MM-dd HH:mm,20} | {row.OrdersPerMinute,7:F0} | {row.Pods,10:F2} | {p.RecommendedPods,9} | {row.Cpu,10:F1} | {p.RecommendedCpuMillisPerPod,8}");
}

Console.WriteLine("\nDone.");
