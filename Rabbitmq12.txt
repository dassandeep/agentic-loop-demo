    // 3. DATABASE LAYER — Order Timestamps Only
    // ============================================================================

    public interface IOrderRepository
    {
        Task<List<DateTime>> GetOrderTimestampsAsync(string ns, int days, CancellationToken ct);
    }

    public class OrderRepository : IOrderRepository
    {
        private readonly string _connectionString;

        public OrderRepository(string connectionString)
        {
            _connectionString = connectionString;
        }

        public async Task<List<DateTime>> GetOrderTimestampsAsync(string ns, int days, CancellationToken ct)
        {
            var results = new List<DateTime>();
            var cutoff = DateTime.UtcNow.AddDays(-days);

            // Production: replace with parameterized query against your actual Orders table.
            // Assumption: table Orders(OrderNo, CreatedAtUtc, Namespace)
            const string sql = @"
                SELECT CreatedAtUtc 
                FROM Orders 
                WHERE Namespace = @Namespace 
                  AND CreatedAtUtc >= @Cutoff
                ORDER BY CreatedAtUtc";

            using var conn = new SqlConnection(_connectionString);
            using var cmd = new SqlCommand(sql, conn);
            cmd.Parameters.AddWithValue("@Namespace", ns);
            cmd.Parameters.AddWithValue("@Cutoff", cutoff);

            await conn.OpenAsync(ct).ConfigureAwait(false);
            using var reader = await cmd.ExecuteReaderAsync(ct).ConfigureAwait(false);

            while (await reader.ReadAsync(ct).ConfigureAwait(false))
            {
                results.Add(reader.GetDateTime(0));
            }

            return results;
        }
    }

    // ============================================================================
    // 4. PROMETHEUS LAYER — Pod Memory Metrics
    // ============================================================================

    public interface IPrometheusClient
    {
        Task<List<PodMemorySample>> QueryPodMemoryAsync(string query, DateTime startUtc, DateTime endUtc, CancellationToken ct);
    }

    public class PodMemorySample
    {
        public DateTime TimestampUtc { get; set; }
        public string PodName { get; set; } = string.Empty;
        public double MemoryBytes { get; set; }
    }

    public class PrometheusClient : IPrometheusClient
    {
        private readonly HttpClient _http;
        private readonly ILogger<PrometheusClient> _logger;

        public PrometheusClient(HttpClient http, ILogger<PrometheusClient> logger)
        {
            _http = http;
            _logger = logger;
        }

        public async Task<List<PodMemorySample>> QueryPodMemoryAsync(
            string query, DateTime startUtc, DateTime endUtc, CancellationToken ct)
        {
            var samples = new List<PodMemorySample>();

            // Prometheus range query API
            var start = startUtc.ToString("o", CultureInfo.InvariantCulture);
            var end = endUtc.ToString("o", CultureInfo.InvariantCulture);

            var url = $"/api/v1/query_range?query={Uri.EscapeDataString(query)}" +
                      $"&start={Uri.EscapeDataString(start)}" +
                      $"&end={Uri.EscapeDataString(end)}" +
                      $"&step=60s";

            _logger.LogInformation("Querying Prometheus: {Url}", url);

            using var response = await _http.GetAsync(url, ct).ConfigureAwait(false);
            response.EnsureSuccessStatusCode();

            var json = await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
            using var doc = JsonDocument.Parse(json);

            var root = doc.RootElement;
            if (root.GetProperty("status").GetString() != "success")
                throw new InvalidOperationException("Prometheus query failed");

            var resultArray = root.GetProperty("data").GetProperty("result");

            foreach (var series in resultArray.EnumerateArray())
            {
                var podName = "unknown";
                if (series.TryGetProperty("metric", out var metric))
                {
                    if (metric.TryGetProperty("pod", out var podEl))
                        podName = podEl.GetString() ?? "unknown";
                    else if (metric.TryGetProperty("pod_name", out var podNameEl))
                        podName = podNameEl.GetString() ?? "unknown";
                }

                if (!series.TryGetProperty("values", out var values)) continue;

                foreach (var point in values.EnumerateArray())
                {
                    var ts = DateTimeOffset.FromUnixTimeSeconds(point[0].GetInt64()).UtcDateTime;
                    var val = point[1].GetString();

                    if (double.TryParse(val, NumberStyles.Float, CultureInfo.InvariantCulture, out var bytes))
                    {
                        samples.Add(new PodMemorySample
                        {
                            TimestampUtc = ts,
                            PodName = podName,
                            MemoryBytes = bytes
                        });
                    }
                }
            }

            return samples;
        }
    }

    // ============================================================================
    // 5. FEATURE ENGINEERING — Align Orders + Memory into Training Rows
    // ============================================================================

    /// <summary>
    /// Represents one training row: a time bucket with order count and observed pod memory.
    /// </summary>
    public class TrainingRow
    {
        public float OrderCount { get; set; }
        public float MemoryGiB { get; set; }
        public float PodCount { get; set; }
        public DateTime BucketStartUtc { get; set; }
    }

    public class TrainingRowWithLabel
    {
        [LoadColumn(0)] public float OrderCount { get; set; }
        [LoadColumn(1)] public float MemoryGiB { get; set; }
        [LoadColumn(2)] public float PodCount { get; set; }
        [LoadColumn(3)] public float Label_PodCount { get; set; } // what we want to predict
    }

    public class ModelPrediction
    {
        [ColumnName("Score")] public float PredictedPodCount { get; set; }
    }

    // ============================================================================
    // 6. ML.NET TRAINING PIPELINE
    // ============================================================================

    public interface IScalingModelService
    {
        Task TrainAsync(List<TrainingRow> rows, CancellationToken ct);
        Task<double> PredictPodCountAsync(float orderCount, float memoryGiB, float podCount);
        Task LoadAsync();
    }

    public class ScalingModelService : IScalingModelService
    {
        private readonly MLContext _mlContext;
        private ITransformer? _model;
        private readonly string _modelPath;
        private readonly ILogger<ScalingModelService> _logger;

        public ScalingModelService(string modelPath, ILogger<ScalingModelService> logger)
        {
            _mlContext = new MLContext(seed: 1);
            _modelPath = modelPath;
            _logger = logger;
        }

        public async Task TrainAsync(List<TrainingRow> rows, CancellationToken ct)
        {
            if (rows.Count < 10)
                throw new InvalidOperationException("Insufficient training data (need at least 10 aligned buckets).");

            var data = rows.Select(r => new TrainingRowWithLabel
            {
                OrderCount = r.OrderCount,
                MemoryGiB = r.MemoryGiB,
                PodCount = r.PodCount,
                Label_PodCount = r.PodCount // label = current pod count; model learns relationship
            }).ToList();

            IDataView dataView = _mlContext.Data.LoadFromEnumerable(data);

            var pipeline = _mlContext.Transforms
                .Concatenate("Features", nameof(TrainingRowWithLabel.OrderCount),
                    nameof(TrainingRowWithLabel.MemoryGiB), nameof(TrainingRowWithLabel.PodCount))
                .Append(_mlContext.Regression.Trainers.Sdca(
                    labelColumnName: nameof(TrainingRowWithLabel.Label_PodCount),
                    maximumNumberOfIterations: 100));

            _logger.LogInformation("Training ML.NET regression model on {Count} rows...", rows.Count);
            _model = pipeline.Fit(dataView);

            // Save model
            _mlContext.Model.Save(_model, dataView.Schema, _modelPath);
            _logger.LogInformation("Model saved to {Path}", _modelPath);
        }

        public async Task LoadAsync()
        {
            if (!File.Exists(_modelPath))
                throw new FileNotFoundException("Model file not found", _modelPath);

            _model = await Task.Run(() =>
            {
                DataViewSchema schema;
                return _mlContext.Model.Load(_modelPath, out schema);
            });
        }

        public async Task<double> PredictPodCountAsync(float orderCount, float memoryGiB, float podCount)
        {
            if (_model == null)
                throw new InvalidOperationException("Model not trained or loaded.");

            var input = new TrainingRowWithLabel
            {
                OrderCount = orderCount,
                MemoryGiB = memoryGiB,
                PodCount = podCount
            };

            var predEngine = _mlContext.Model.CreatePredictionEngine<TrainingRowWithLabel, ModelPrediction>(_model);
            var result = predEngine.Predict(input);

            return await Task.FromResult(Math.Max(1, Math.Round(result.PredictedPodCount)));
        }
    }

    // ============================================================================
    // 7. ORCHESTRATOR — The Main Logic
    // ============================================================================

    public interface IScalingOrchestrator
    {
        Task<ScalingResponse> ComputeScalingAsync(ScalingRequest request, CancellationToken ct);
    }

    public class ScalingOrchestrator : IScalingOrchestrator
    {
        private readonly IOrderRepository _orderRepo;
        private readonly IPrometheusClient _promClient;
        private readonly IScalingModelService _modelService;
        private readonly AppConfig _config;
        private readonly ILogger<ScalingOrchestrator> _logger;

        public ScalingOrchestrator(
            IOrderRepository orderRepo,
            IPrometheusClient promClient,
            IScalingModelService modelService,
            AppConfig config,
            ILogger<ScalingOrchestrator> logger)
        {
            _orderRepo = orderRepo;
            _promClient = promClient;
            _modelService = modelService;
            _config = config;
            _logger = logger;
        }

        public async Task<ScalingResponse> ComputeScalingAsync(ScalingRequest req, CancellationToken ct)
        {
            _logger.LogInformation("Computing scaling for {Namespace}...", req.Namespace);

            var now = DateTime.UtcNow;
            var windowStart = now.AddMinutes(-req.WithinZMinutes);
            var trainingStart = now.AddDays(-req.Days);

            // ---------------------------------------------------------------
            // 7a. Fetch raw data in parallel
            // ---------------------------------------------------------------
            var ordersTask = _orderRepo.GetOrderTimestampsAsync(req.Namespace, req.Days, ct);
            var memoryTask = _promClient.QueryPodMemoryAsync(_config.PrometheusQuery, trainingStart, now, ct);

            await Task.WhenAll(ordersTask, memoryTask).ConfigureAwait(false);

            var orderTimestamps = ordersTask.Result;
            var memorySamples = memoryTask.Result;

            _logger.LogInformation("Fetched {Orders} orders and {Samples} memory samples.",
                orderTimestamps.Count, memorySamples.Count);

            // ---------------------------------------------------------------
            // 7b. Build time-bucketed training rows (5-min buckets)
            // ---------------------------------------------------------------
            const int bucketMinutes = 5;
            var trainingRows = BuildTrainingRows(orderTimestamps, memorySamples, trainingStart, now, bucketMinutes);

            // ---------------------------------------------------------------
            // 7c. Train or load model
            // ---------------------------------------------------------------
            try
            {
                await _modelService.TrainAsync(trainingRows, ct).ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "Training failed, attempting to load existing model.");
                await _modelService.LoadAsync().ConfigureAwait(false);
            }

            // ---------------------------------------------------------------
            // 7d. Compute target demand metrics
            // ---------------------------------------------------------------

            // Count orders in the recent Z-minute window
            var recentOrders = orderTimestamps.Count(t => t >= windowStart && t <= now);

            // Scale up to the targetOrders if recent is lower (target is the goal)
            var effectiveTargetOrders = Math.Max(req.TargetOrders, recentOrders);

            // Derive average memory per pod from recent Prometheus samples
            var recentMemory = memorySamples.Where(s => s.TimestampUtc >= windowStart).ToList();
            double avgMemoryPerPodBytes = recentMemory.Count > 0
                ? recentMemory.Average(s => s.MemoryBytes)
                : 0;

            double avgMemoryPerPodGiB = avgMemoryPerPodBytes / (1024.0 * 1024.0 * 1024.0);

            // Current pod count (distinct pods in recent window)
            double currentPodCount = recentMemory.Count > 0
                ? recentMemory.Select(s => s.PodName).Distinct().Count()
                : 1;

            // ---------------------------------------------------------------
            // 7e. Predict pod count using ML.NET
            // ---------------------------------------------------------------
            // Feed the model: expected orders in window, avg memory per pod, current pod count
            double predictedPods = await _modelService.PredictPodCountAsync(
                effectiveTargetOrders,
                (float)avgMemoryPerPodGiB,
                (float)currentPodCount).ConfigureAwait(false);

            // ---------------------------------------------------------------
            // 7f. Apply basis adjustment (MAX vs AVG)
            // ---------------------------------------------------------------
            if (req.Basis.Equals("MAX", StringComparison.OrdinalIgnoreCase))
            {
                // Use peak order pressure — approximate peak as effectiveTargetOrders / window fraction
                var peakFactor = 1.0 + (req.HeadroomZPercent / 100.0);
                predictedPods = Math.Ceiling(predictedPods * peakFactor);
            }
            else if (req.Basis.Equals("AVG", StringComparison.OrdinalIgnoreCase))
            {
                // Use average only — no peak amplification
                _logger.LogInformation("Basis=AVG: using average order pressure.");
            }
            else
            {
                _logger.LogWarning("Unknown basis '{Basis}', defaulting to MAX.", req.Basis);
                predictedPods = Math.Ceiling(predictedPods * (1.0 + req.HeadroomZPercent / 100.0));
            }

            // Apply explicit headroom if basis is MAX but headroom not already applied
            if (!req.Basis.Equals("MAX", StringComparison.OrdinalIgnoreCase))
            {
                predictedPods = Math.Ceiling(predictedPods * (1.0 + req.HeadroomZPercent / 100.0));
            }

            // ---------------------------------------------------------------
            // 7g. Compute recommended memory
            // ---------------------------------------------------------------
            // Memory scales with pod count; use observed per-pod average as baseline
            double recommendedMemoryGiB = avgMemoryPerPodGiB * predictedPods;

            // Fallback: if no memory data, derive a sensible default
            if (recommendedMemoryGiB <= 0)
            {
                recommendedMemoryGiB = predictedPods * 0.5; // 512 MiB per pod fallback
            }

            return new ScalingResponse
            {
                Success = true,
                Message = $"Basis={req.Basis}, TargetOrders={effectiveTargetOrders}, Window={req.WithinZMinutes}min",
                RecommendedPods = (int)predictedPods,
                RecommendedMemoryGiB = Math.Round(recommendedMemoryGiB, 2),
                CurrentPodCount = currentPodCount,
                CurrentMemoryGiB = Math.Round(avgMemoryPerPodGiB, 2),
                PredictedOrdersInWindow = effectiveTargetOrders,
                PredictedMemoryPerPodGiB = Math.Round(avgMemoryPerPodGiB, 3),
                GeneratedAtUtc = now
            };
        }

        // -------------------------------------------------------------------
        // Feature engineering: align orders and memory into 5-min buckets
        // -------------------------------------------------------------------
        private static List<TrainingRow> BuildTrainingRows(
            List<DateTime> orderTimestamps,
            List<PodMemorySample> memorySamples,
            DateTime start,
            DateTime end,
            int bucketMinutes)
        {
            var rows = new List<TrainingRow>();

            // Aggregate orders into buckets
            var orderBuckets = orderTimestamps
                .Where(t => t >= start && t <= end)
                .GroupBy(t => TruncateToBucket(t, bucketMinutes))
                .ToDictionary(g => g.Key, g => g.Count());

            // Aggregate memory into buckets (sum across pods per bucket)
            var memoryBuckets = memorySamples
                .Where(s => s.TimestampUtc >= start && s.TimestampUtc <= end)
                .GroupBy(s => TruncateToBucket(s.TimestampUtc, bucketMinutes))
                .ToDictionary(g => g.Key, g => new
                {
                    TotalBytes = g.Sum(x => x.MemoryBytes),
                    PodCount = g.Select(x => x.PodName).Distinct().Count()
                });

            // Union of bucket keys
            var allBuckets = orderBuckets.Keys.Union(memoryBuckets.Keys).OrderBy(k => k).ToList();

            foreach (var bucket in allBuckets)
            {
                var orderCount = orderBuckets.GetValueOrDefault(bucket, 0);
                var memInfo = memoryBuckets.GetValueOrDefault(bucket);

                if (memInfo == null) continue; // skip buckets with no memory data

                var memoryGiB = memInfo.TotalBytes / (1024.0 * 1024.0 * 1024.0);

                rows.Add(new TrainingRow
                {
                    OrderCount = orderCount,
                    MemoryGiB = (float)memoryGiB,
                    PodCount = memInfo.PodCount,
                    BucketStartUtc = bucket
                });
            }

            return rows;
        }

        private static DateTime TruncateToBucket(DateTime dt, int minutes)
        {
            var ticks = TimeSpan.FromMinutes(minutes).Ticks;
            return new DateTime(dt.Ticks - (dt.Ticks % ticks), DateTimeKind.Utc);
        }
    }

    // ============================================================================
    // 8. API HOST — Production entry point
    // ============================================================================

    public class Program
    {
        public static async Task Main(string[] args)
        {
            var host = Host.CreateDefaultBuilder(args)
                .ConfigureServices((context, services) =>
                {
                    // Configuration — bind from env vars or appsettings
                    var config = new AppConfig
                    {
                        ConnectionString = Environment.GetEnvironmentVariable("DB_CONNECTION")
                            ?? "Server=localhost;Database=Orders;Trusted_Connection=True;",
                        PrometheusBaseUrl = Environment.GetEnvironmentVariable("PROMETHEUS_URL")
                            ?? "http://localhost:9090"
                    };

                    services.AddSingleton(config);

                    // HTTP client for Prometheus
                    services.AddHttpClient<IPrometheusClient, PrometheusClient>(client =>
                    {
                        client.BaseAddress = new Uri(config.PrometheusBaseUrl);
                        client.Timeout = TimeSpan.FromSeconds(30);
                    });

                    // Repositories
                    services.AddSingleton<IOrderRepository>(sp =>
                        new OrderRepository(config.ConnectionString));

                    // ML model service (singleton to avoid retraining per request)
                    services.AddSingleton<IScalingModelService>(sp =>
                        new ScalingModelService(
                            config.MlModelPath,
                            sp.GetRequiredService<ILogger<ScalingModelService>>()));

                    // Orchestrator
                    services.AddSingleton<IScalingOrchestrator, ScalingOrchestrator>();
                })
                .Build();

            await host.StartAsync();

            // DEMO: Simulate a request
            var orchestrator = host.Services.GetRequiredService<IScalingOrchestrator>();
            var logger = host.Services.GetRequiredService<ILogger<Program>>();

            var request = new ScalingRequest
            {
                Namespace = "gcp-pref-customer",
                TargetOrders = 10000,
                WithinZMinutes = 30,
                Days = 15,
                HeadroomZPercent = 20,
                Basis = "MAX"
            };

            try
            {
                var result = await orchestrator.ComputeScalingAsync(request, CancellationToken.None);
                logger.LogInformation("=== SCALING RECOMMENDATION ===");
                logger.LogInformation("Namespace:            {Namespace}", request.Namespace);
                logger.LogInformation("Recommended Pods:     {Pods}", result.RecommendedPods);
                logger.LogInformation("Recommended Memory:   {Memory} GiB", result.RecommendedMemoryGiB);
                logger.LogInformation("Current Pods:         {CurrentPods}", result.CurrentPodCount);
                logger.LogInformation("Current Mem/Pod:      {CurrentMem} GiB", result.CurrentMemoryGiB);
                logger.LogInformation("Message:              {Message}", result.Message);
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Scaling computation failed.");
            }

            await host.StopAsync();
        }
    }
