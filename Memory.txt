public static class MemoryProbe
{
    public static void ForceGC()
    {
        GC.Collect();
        GC.WaitForPendingFinalizers();
        GC.Collect();
    }

    public static long ManagedHeap() => GC.GetTotalMemory(forceFullCollection: false);
    public static long TotalAllocated() => GC.GetTotalAllocatedBytes(precise: false);
    public static long WorkingSet() => Process.GetCurrentProcess().WorkingSet64;

    public sealed class Snapshot
    {
        public long ManagedHeap;
        public long TotalAllocated;
        public long WorkingSet;
    }

    public static Snapshot Take()
    {
        ForceGC();
        return new Snapshot
        {
            ManagedHeap = ManagedHeap(),
            TotalAllocated = TotalAllocated(),
            WorkingSet = WorkingSet()
        };
    }

    public static void Report(string label, Snapshot before, int recordCount = 0)
    {
        var after = new Snapshot
        {
            ManagedHeap = ManagedHeap(),
            TotalAllocated = TotalAllocated(),
            WorkingSet = WorkingSet()
        };

        Console.WriteLine($"── {label} ──");
        if (recordCount > 0)
        {
            Console.WriteLine($"   Records processed:     {recordCount:N0}");
            Console.WriteLine($"   ~bytes / record:       {(after.ManagedHeap - before.ManagedHeap) / (double)recordCount:F1} (live heap delta)");
        }
        Console.WriteLine($"   Live managed heap Δ:   {FmtDelta(after.ManagedHeap - before.ManagedHeap)}   (now {Fmt(after.ManagedHeap)})");
        Console.WriteLine($"   Total allocated Δ:     {FmtDelta(after.TotalAllocated - before.TotalAllocated)}");
        Console.WriteLine($"   Working set Δ:         {FmtDelta(after.WorkingSet - before.WorkingSet)}   (now {Fmt(after.WorkingSet)})");
        Console.WriteLine($"   GC gen0/1/2 counts:    {GC.CollectionCount(0)}/{GC.CollectionCount(1)}/{GC.CollectionCount(2)}");
        Console.WriteLine();
    }

    private static string Fmt(long bytes) => bytes switch
    {
        >= 1024L * 1024 * 1024 => $"{bytes / 1024.0 / 1024 / 1024:F2} GB",
        >= 1024L * 1024        => $"{bytes / 1024.0 / 1024:F2} MB",
        >= 1024L                => $"{bytes / 1024.0:F2} KB",
        _                       => $"{bytes} B"
    };

    private static string FmtDelta(long bytes) => (bytes >= 0 ? "+" : "-") + Fmt(Math.Abs(bytes));
}

static void InsertDemo(FulfillmentLineFileStore store, int count)
{
    if (store.Exists())
    {
        Console.Write($"File exists ({MemoryProbe.FormatBytes(store.FileSizeBytes())}). Overwrite? [y/N] ");
        if (!Console.ReadLine()?.Equals("y", StringComparison.OrdinalIgnoreCase) ?? true) return;
        store.Delete();
    }

    Console.WriteLine($"Generating and inserting {count:N0} records in batches of {BatchSize:N0}…");
    var before = MemoryProbe.Take();
    var sw = Stopwatch.StartNew();

    for (int i = 0; i < count; i += BatchSize)
    {
        var batch = new List<FulfillmentLine>(BatchSize);
        for (int j = 0; j < BatchSize && i + j < count; j++)
        {
            int idx = i + j;
            batch.Add(new FulfillmentLine
            {
                FulfillmentLineId = $"FL-{idx:D8}",
                FulfillmentId = $"F-{idx / LinesPerFulfillment:D4}",
                OrderLineId = $"OL-{idx:D8}",
                Quantity = 1 + (idx % 10)
            });
        }
        store.Append(batch);
    }

    sw.Stop();
    MemoryProbe.Report($"Insert {count:N0} records", before, count);
    Console.WriteLine($"   Elapsed: {sw.Elapsed.TotalSeconds:F2}s");
    Console.WriteLine($"   File size: {MemoryProbe.FormatBytes(store.FileSizeBytes())}");
    Console.WriteLine();
}

// FILE STORE — NDJSON, one FulfillmentLine per line
// ══════════════════════════════════════════════════════════════

public sealed class FulfillmentLineFileStore
{
    private readonly string _path;
    private readonly JsonSerializerOptions _json = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
    };

    public FulfillmentLineFileStore(string path) => _path = path;

    public bool Exists() => File.Exists(_path);
    public long FileSizeBytes() => Exists() ? new FileInfo(_path).Length : 0;
    public void Delete() { if (Exists()) File.Delete(_path); }

    public void Append(IReadOnlyList<FulfillmentLine> batch)
    {
        using var fs = new FileStream(_path, FileMode.Append, FileAccess.Write, FileShare.Read, bufferSize: 64 * 1024);
        using var sw = new StreamWriter(fs, new UTF8Encoding(false), bufferSize: 64 * 1024);
        foreach (var line in batch)
            sw.WriteLine(JsonSerializer.Serialize(line, _json));
    }

    // Streaming: one object alive at a time.
    public IEnumerable<FulfillmentLine> StreamAll()
    {
        using var fs = new FileStream(_path, FileMode.Open, FileAccess.Read, FileShare.Read, bufferSize: 64 * 1024);
        using var sr = new StreamReader(fs, Encoding.UTF8, false, bufferSize: 64 * 1024);
        string? raw;
        while ((raw = sr.ReadLine()) is not null)
        {
            if (raw.Length == 0) continue;
            yield return JsonSerializer.Deserialize<FulfillmentLine>(raw, _json)!;
        }
    }

    public IEnumerable<FulfillmentLine> StreamByFulfillment(string fulfillmentId)
    {
        foreach (var line in StreamAll())
            if (string.Equals(line.FulfillmentId, fulfillmentId, StringComparison.Ordinal))
                yield return line;
    }

    // Naive: entire file into a string, split, then materialize.
    public List<FulfillmentLine> LoadAllNaive()
    {
        var text = File.ReadAllText(_path);
        var parts = text.Split('\n');
        var result = new List<FulfillmentLine>(parts.Length);
        foreach (var raw in parts)
        {
            var trimmed = raw.Trim();
            if (trimmed.Length == 0) continue;
            result.Add(JsonSerializer.Deserialize<FulfillmentLine>(trimmed, _json)!);
        }
        return result;
    }
}

public static class MemoryProbe
{
    public sealed record Snapshot(long ManagedHeap, long TotalAllocated, long WorkingSet);

    public static void ForceGC()
    {
        GC.Collect();
        GC.WaitForPendingFinalizers();
        GC.Collect();
    }

    public static Snapshot Take()
    {
        ForceGC();
        return new Snapshot(
            GC.GetTotalMemory(false),
            GC.GetTotalAllocatedBytes(false),
            Process.GetCurrentProcess().WorkingSet64);
    }

    public static void Report(string label, Snapshot before, int recordCount = 0)
    {
        var managed = GC.GetTotalMemory(false);
        var allocated = GC.GetTotalAllocatedBytes(false);
        var working = Process.GetCurrentProcess().WorkingSet64;

        Console.WriteLine($"── {label} ──");
        if (recordCount > 0)
        {
            Console.WriteLine($"   Records:                 {recordCount:N0}");
            Console.WriteLine($"   ~bytes/record (live):    {(managed - before.ManagedHeap) / (double)recordCount:F0}");
        }
        Console.WriteLine($"   Live managed heap Δ:     {Delta(managed - before.ManagedHeap),14}   (now {FormatBytes(managed)})");
        Console.WriteLine($"   Total allocated Δ:       {Delta(allocated - before.TotalAllocated),14}   (cumulative work, mostly garbage)");
        Console.WriteLine($"   Working set Δ:           {Delta(working - before.WorkingSet),14}   (now {FormatBytes(working)})");
        Console.WriteLine($"   GC gen0/1/2 counts:      {GC.CollectionCount(0)}/{GC.CollectionCount(1)}/{GC.CollectionCount(2)}");
        Console.WriteLine();
    }

    public static string FormatBytes(long bytes)
    {
        double abs = Math.Abs(bytes);
        string unit = abs >= 1L << 30 ? "GB" : abs >= 1L << 20 ? "MB" : abs >= 1L << 10 ? "KB" : "B";
        double divisor = unit switch { "GB" => 1 << 30, "MB" => 1 << 20, "KB" => 1 << 10, _ => 1 };
        return $"{bytes / divisor:F2} {unit}";
    }

    private static string Delta(long bytes) => (bytes >= 0 ? "+" : "-") + FormatBytes(Math.Abs(bytes));
}

    private static void InsertDemo(FulfillmentLineFileStore store)
    {
        if (store.Exists())
        {
            Console.Write($"File exists ({MemoryProbe.FormatBytes(store.FileSizeBytes())}). Overwrite? [y/N]: ");
            if (!string.Equals(Console.ReadLine()?.Trim(), "y", StringComparison.OrdinalIgnoreCase)) return;
            store.Delete();
        }

        Console.Write($"How many records? [{DefaultCount:N0}]: ");
        var raw = Console.ReadLine()?.Trim();
        int count = string.IsNullOrEmpty(raw) ? DefaultCount : (int.TryParse(raw, out var n) && n > 0 ? n : DefaultCount);

        Console.WriteLine($"\nInserting {count:N0} records in batches of {BatchSize:N0}…");

        var before = MemoryProbe.Take();
        var sw = Stopwatch.StartNew();

        for (int i = 0; i < count; i += BatchSize)
        {
            var batch = new List<FulfillmentLine>(BatchSize);
            for (int j = 0; j < BatchSize && i + j < count; j++)
            {
                int idx = i + j;
                batch.Add(new FulfillmentLine
                {
                    FulfillmentLineId = $"FL-{idx:D8}",
                    FulfillmentId = $"F-{idx / LinesPerFulfillment:D4}",
                    OrderLineId = $"OL-{idx:D8}",
                    Quantity = 1 + (idx % 10)
                });

            store.Append(batch);
        }

        sw.Stop();
        MemoryProbe.Report($"Insert {count:N0} records (batched streaming write)", before, count);
        Console.WriteLine($"   Elapsed:   {sw.Elapsed.TotalSeconds:F2}s");

    private static void NaiveRetrieve(FulfillmentLineFileStore store)
    {
        if (!EnsureData(store)) return;
        Console.WriteLine("❌  NAIVE retrieval: File.ReadAllText → Split → List<T>");
        var before = MemoryProbe.Take();
        var sw = Stopwatch.StartNew();

        var all = store.LoadAllNaive();
        long sum = 0;
        foreach (var l in all) sum += l.Quantity;

        sw.Stop();
        MemoryProbe.Report("Naive load ALL", before, all.Count);
        Console.WriteLine($"   Elapsed: {sw.Elapsed.TotalSeconds:F2}s   Sum(Quantity)={sum:N0}");
        Console.WriteLine("   ⚠ objects are still alive at report time — that's the live-heap cost.");
        Console.WriteLine();
        GC.KeepAlive(all);
    }

    private static void StreamingRetrieve(FulfillmentLineFileStore store)
    {
        if (!EnsureData(store)) return;
        Console.WriteLine("✅  STREAMING retrieval: StreamReader.ReadLine → yield");

        int count = 0;
        long sum = 0;
        foreach (var l in store.StreamAll())
        {
            count++;
            sum += l.Quantity;
        }

        sw.Stop();
        MemoryProbe.Report("Streaming iteration", before, count);
        Console.WriteLine($"   Elapsed: {sw.Elapsed.TotalSeconds:F2}s   Sum(Quantity)={sum:N0}");

// 2. FILE STORE — NDJSON, one FulfillmentLine per line
// ============================================================================

public sealed class FulfillmentLineFileStore
{
    private readonly string _path;
    private readonly JsonSerializerOptions _json = new()
    {
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase,
        DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull
    };

    public FulfillmentLineFileStore(string path) => _path = path;

    public bool Exists() => File.Exists(_path);
    public long FileSizeBytes() => Exists() ? new FileInfo(_path).Length : 0;
    public void Delete() { if (Exists()) File.Delete(_path); }

    // Batched append: never more than one batch alive.
    public void Append(IReadOnlyList<FulfillmentLine> batch)
    {
        using var fs = new FileStream(_path, FileMode.Append, FileAccess.Write, FileShare.Read, bufferSize: 64 * 1024);
        using var sw = new StreamWriter(fs, new UTF8Encoding(false), bufferSize: 64 * 1024);
        foreach (var line in batch)
            sw.WriteLine(JsonSerializer.Serialize(line, _json));
    }

    // Streaming: one object alive at a time.
    public IEnumerable<FulfillmentLine> StreamAll()
    {
        using var fs = new FileStream(_path, FileMode.Open, FileAccess.Read, FileShare.Read, bufferSize: 64 * 1024);
        using var sr = new StreamReader(fs, Encoding.UTF8, detectEncodingFromByteOrderMarks: false, bufferSize: 64 * 1024);
        string? raw;
        while ((raw = sr.ReadLine()) is not null)
        {
            if (raw.Length == 0) continue;
            yield return JsonSerializer.Deserialize<FulfillmentLine>(raw, _json)!;
        }
    }

    public IEnumerable<FulfillmentLine> StreamByFulfillment(string fulfillmentId)
    {
        foreach (var line in StreamAll())
            if (string.Equals(line.FulfillmentId, fulfillmentId, StringComparison.Ordinal))
                yield return line;
    }

    // Naive: whole file into a string, split, then materialize every entity.
    public List<FulfillmentLine> LoadAllNaive()
    {
        var text  = File.ReadAllText(_path);
        var parts = text.Split('\n');
        var result = new List<FulfillmentLine>(parts.Length);
        foreach (var raw in parts)
        {
            var trimmed = raw.Trim();
            if (trimmed.Length == 0) continue;
            result.Add(JsonSerializer.Deserialize<FulfillmentLine>(trimmed, _json)!);
        }
        return result;
    }
}

public static class MemoryProbe
{
    public sealed class Snapshot
    {
        public long ManagedHeap { get; init; }
        public long TotalAllocated { get; init; }
        public long WorkingSet { get; init; }
    }

    public static void ForceGC()
    {
        GC.Collect();
        GC.WaitForPendingFinalizers();
        GC.Collect();
    }

    public static Snapshot Take()
    {
        ForceGC();
        return new Snapshot
        {
            ManagedHeap    = GC.GetTotalMemory(false),
            TotalAllocated = GC.GetTotalAllocatedBytes(false),
            WorkingSet     = Process.GetCurrentProcess().WorkingSet64
        };
    }

    public static void Report(string label, Snapshot before, int recordCount = 0)
    {
        var managed   = GC.GetTotalMemory(false);
        var allocated = GC.GetTotalAllocatedBytes(false);
        var working   = Process.GetCurrentProcess().WorkingSet64;

        Console.WriteLine($"-- {label} --");
        if (recordCount > 0)
        {
            Console.WriteLine($"   Records processed      : {recordCount:N0}");
            Console.WriteLine($"   ~bytes / record (live) : {(managed - before.ManagedHeap) / (double)recordCount:F0}");
        }
        Console.WriteLine($"   Live managed heap d    : {Delta(managed - before.ManagedHeap)}   (now {FormatBytes(managed)})");
        Console.WriteLine($"   Total allocated d      : {Delta(allocated - before.TotalAllocated)}   (cumulative; mostly transient)");
        Console.WriteLine($"   Working set d          : {Delta(working - before.WorkingSet)}   (now {FormatBytes(working)})");
        Console.WriteLine($"   GC gen0/1/2 counts     : {GC.CollectionCount(0)}/{GC.CollectionCount(1)}/{GC.CollectionCount(2)}");
        Console.WriteLine();
    }

    public static string FormatBytes(long bytes)
    {
        double abs = Math.Abs(bytes);
        string unit = abs >= (1L << 30) ? "GB" : abs >= (1L << 20) ? "MB" : abs >= (1L << 10) ? "KB" : "B";
        double divisor = unit switch { "GB" => 1L << 30, "MB" => 1L << 20, "KB" => 1L << 10, _ => 1.0 };
        return $"{bytes / divisor:F2} {unit}";
    }

    private static string Delta(long bytes) => (bytes >= 0 ? "+" : "-") + FormatBytes(Math.Abs(bytes));
}

CLOB :
Doubled Storage:
Oracle internally stores CLOB data using the UCS2 character set, where each character requires two bytes.
 Since JSON is text, this can effectively double the storage footprint for many documents.
 
 Character Set Conversion: JSON is natively UTF-8. Storing it in a CLOB forces the database to perform character-set
 conversion on write and read, adding CPU overhead.
 
 BLOB:
 No Conversion Overhead: A BLOB stores the JSON bytes exactly as they are, avoiding the character-set conversion
 and the 2-byte-per-character penalty of CLOB.

More Efficient: Oracle's documentation explicitly recommends BLOB over CLOB for JSON storage when using LOBs.

JSON :
It uses a specialized binary format called OSON (Oracle Serialized Object Notation).

With CLOB or BLOB, the database must parse the JSON text every time it is queried. The native JSON type stores data
 in a pre-parsed binary format, eliminating this costly step on every access.
 
 The OSON format is more compact than the raw text or binary JSON, leading to the significant storage savings seen in the
 benchmark.
 
 Faster Queries: Because the data is pre-parsed and stored efficiently, query execution is faster.
 Oracle's own materials cite benchmarks showing up to 4.4x higher throughput and up to 13x faster 
 queries with the native JSON type compared to older storage methods.

        if (await reader.ReadAsync())

            var buffer = new char[82_000];       // ~80 KB buffer
            long dataIndex = 0;
            while (true)
            {
                // GetChars(index, dataOffset, buffer, bufferOffset, length)
                var read = reader.GetChars(0, dataIndex, buffer, 0, buffer.Length);
                if (read == 0) break;

                totalChars += read;
                dataIndex += read;
                // In a real pipeline you would process the buffer here
                // (e.g., feed it to a JsonReader or a file stream).
            }
        }

        MemoryProbe.Report("Streaming CLOB (SequentialAccess)", before, 1);
        Console.WriteLine($"   Total characters read: {totalChars:N0}");

using System.Diagnostics;

public static class MemoryProbe
{
    public sealed record Snapshot(long ManagedHeap, long WorkingSet);

    public static void ForceGC()
    {
        GC.Collect();
        GC.WaitForPendingFinalizers();
        GC.Collect();
    }

    public static Snapshot Take() => new(
        GC.GetTotalMemory(false),
        Process.GetCurrentProcess().WorkingSet64);

    public static void Report(string label, Snapshot before, int count)
    {
        ForceGC();
        var heap = GC.GetTotalMemory(false);
        var ws = Process.GetCurrentProcess().WorkingSet64;
        var dHeap = heap - before.ManagedHeap;
        var dWs = ws - before.WorkingSet;

        Console.WriteLine($"-- {label} --");
        if (count > 0)
            Console.WriteLine($"   Records          : {count}");
        Console.WriteLine($"   Δ Managed heap   : {Format(dHeap)} (now {Format(heap)})");
        Console.WriteLine($"   Δ Working set    : {Format(dWs)} (now {Format(ws)})");
        Console.WriteLine();
    }

    public static string FormatBytes(long b) => Format(b);
    private static string Format(long b) =>
        Math.Abs(b) >= 1 << 20 ? $"{b / (double)(1 << 20):F2} MB" :
        Math.Abs(b) >= 1 << 10 ? $"{b / (double)(1 << 10):F2} KB" : $"{b} B";
}
