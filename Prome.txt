// Services/PrometheusQueryService.cs
using Microsoft.Extensions.Logging;
using System;
using System.Collections.Generic;
using System.Linq;

using System.Text.Json;
using System.Threading;
using System.Threading.Tasks;

namespace CSharpK8sController.Services

    public class PrometheusQueryService : IPrometheusQueryService
    {
        private readonly HttpClient _httpClient;
        private readonly ILogger<PrometheusQueryService> _logger;
        private readonly string _prometheusUrl;
        private readonly string _namespace;
        private readonly string _deploymentName;
        private const string PROMETHEUS_QUERY_ENDPOINT = "/api/v1/query";
        private const string PROMETHEUS_QUERY_RANGE_ENDPOINT = "/api/v1/query_range";

        public PrometheusQueryService(
            HttpClient httpClient,
            ILogger<PrometheusQueryService> logger,
            string prometheusUrl = "http://prometheus-operated.monitoring:9090",
            string @namespace = "your-namespace",
            string deploymentName = "csharp-order-app")
        {
            _httpClient = httpClient;
            _logger = logger;
            _prometheusUrl = prometheusUrl.TrimEnd('/');
            _namespace = @namespace;
            _deploymentName = deploymentName;
        }

        /// <summary>
        /// Execute instant query
        /// </summary>
        public async Task<PrometheusQueryResult> QueryAsync(string query, CancellationToken ct = default)
        {
            try
            {
                var url = $"{_prometheusUrl}{PROMETHEUS_QUERY_ENDPOINT}?query={Uri.EscapeDataString(query)}";
                var response = await _httpClient.GetAsync(url, ct);
                response.EnsureSuccessStatusCode();

                var json = await response.Content.ReadAsStringAsync(ct);
                return JsonSerializer.Deserialize<PrometheusQueryResult>(json);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to execute Prometheus query: {Query}", query);
                throw;
            }
        }

        /// <summary>
        /// Execute range query for historical data
        /// </summary>
        public async Task<PrometheusQueryResult> QueryRangeAsync(
            string query,
            DateTime start,
            DateTime end,
            string step = "5m",
            CancellationToken ct = default)
        {
            try
            {
                var startUnix = new DateTimeOffset(start).ToUnixTimeSeconds();
                var endUnix = new DateTimeOffset(end).ToUnixTimeSeconds();

                var url = $"{_prometheusUrl}{PROMETHEUS_QUERY_RANGE_ENDPOINT}?" +
                         $"query={Uri.EscapeDataString(query)}&" +
                         $"start={startUnix}&" +
                         $"end={endUnix}&" +
                         $"step={step}";

                var response = await _httpClient.GetAsync(url, ct);
                response.EnsureSuccessStatusCode();

                var json = await response.Content.ReadAsStringAsync(ct);
                return JsonSerializer.Deserialize<PrometheusQueryResult>(json);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to execute Prometheus range query: {Query}", query);
                throw;
            }
        }

        /// <summary>
        /// Get historical pod metrics for a time range
        /// </summary>
        public async Task<List<PodMetricHistory>> GetHistoricalPodMetricsAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            try
            {
                _logger.LogInformation($"📊 Fetching historical metrics from {start} to {end}");

                var results = new List<PodMetricHistory>();

                // 1. Get CPU usage history
                var cpuQuery = $@"
                    sum(
                        rate(
                            container_cpu_usage_seconds_total{{
                                namespace=""{_namespace}"",
                                pod=~""{_deploymentName}.*"",
                                container!=""
                            }}[5m]
                        )
                    ) by (pod)";

                var cpuResult = await QueryRangeAsync(cpuQuery, start, end, "5m", ct);

                // 2. Get Memory usage history
                var memoryQuery = $@"
                    sum(
                        container_memory_working_set_bytes{{
                            namespace=""{_namespace}"",
                            pod=~""{_deploymentName}.*"",
                            container!=""
                        }}
                    ) by (pod)";

                var memoryResult = await QueryRangeAsync(memoryQuery, start, end, "5m", ct);

                // 3. Get Pod count history
                var podCountQuery = $@"
                    count(
                        kube_pod_info{{
                            namespace=""{_namespace}"",
                            pod=~""{_deploymentName}.*""
                        }}
                    )";

                var podCountResult = await QueryRangeAsync(podCountQuery, start, end, "5m", ct);

                // 4. Correlate the data
                var timestamps = new HashSet<DateTime>();
                
                // Collect all timestamps
                foreach (var result in cpuResult.Data.Result)
                {
                    foreach (var value in result.Values)
                    {
                        timestamps.Add(DateTimeOffset.FromUnixTimeSeconds((long)value[0]).DateTime);
                    }
                }

                // Build pod metrics
                foreach (var timestamp in timestamps)
                {
                    var cpuData = cpuResult.Data.Result
                        .SelectMany(r => r.Values
                            .Where(v => DateTimeOffset.FromUnixTimeSeconds((long)v[0]).DateTime == timestamp)
                            .Select(v => new
                            {
                                Pod = r.Metric.GetValueOrDefault("pod", "unknown"),
                                Value = v[1]
                            }));

                    var memoryData = memoryResult.Data.Result
                        .SelectMany(r => r.Values
                            .Where(v => DateTimeOffset.FromUnixTimeSeconds((long)v[0]).DateTime == timestamp)
                            .Select(v => new
                            {
                                Pod = r.Metric.GetValueOrDefault("pod", "unknown"),
                                Value = v[1]
                            }));

                    var podCount = podCountResult.Data.Result
                        .SelectMany(r => r.Values
                            .Where(v => DateTimeOffset.FromUnixTimeSeconds((long)v[0]).DateTime == timestamp))
                        .Select(v => (int)v[1])
                        .FirstOrDefault();

                    // Combine CPU and Memory data
                    var pods = cpuData
                        .Join(memoryData,
                            cpu => cpu.Pod,
                            mem => mem.Pod,
                            (cpu, mem) => new PodMetricHistory
                            {
                                Timestamp = timestamp,
                                PodName = cpu.Pod,
                                CpuUsage = cpu.Value,
                                MemoryUsage = mem.Value,
                                ReplicaCount = podCount
                            })
                        .ToList();

                    results.AddRange(pods);
                }

                _logger.LogInformation($"✅ Retrieved {results.Count} historical metrics entries");
                return results;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to get historical pod metrics");
                throw;
            }
        }

        /// <summary>
        /// Get pod metrics at a specific timestamp
        /// </summary>
        public async Task<PodMetricHistory> GetPodMetricsAtTimeAsync(
            DateTime timestamp,
            CancellationToken ct = default)
        {
            try
            {
                var start = timestamp.AddMinutes(-5);
                var end = timestamp.AddMinutes(5);

                var metrics = await GetHistoricalPodMetricsAsync(start, end, ct);
                
                // Find closest timestamp
                var closest = metrics
                    .GroupBy(m => m.Timestamp)
                    .Select(g => new
                    {
                        Timestamp = g.Key,
                        AvgCpu = g.Average(m => m.CpuUsage),
                        AvgMemory = g.Average(m => m.MemoryUsage),
                        Replicas = g.First().ReplicaCount,
                        TimeDiff = Math.Abs((g.Key - timestamp).TotalSeconds)
                    })
                    .OrderBy(x => x.TimeDiff)
                    .FirstOrDefault();

                if (closest == null)
                    return null;

                return new PodMetricHistory
                {
                    Timestamp = closest.Timestamp,
                    PodName = "average",
                    CpuUsage = closest.AvgCpu,
                    MemoryUsage = closest.AvgMemory,
                    ReplicaCount = closest.Replicas
                };
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to get pod metrics at time {Timestamp}", timestamp);
                throw;
            }
        }

        /// <summary>
        /// Get correlated metrics with order volume
        /// </summary>
        public async Task<List<PodMetricsWithVolume>> GetCorrelatedMetricsAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            try
            {
                _logger.LogInformation("🔄 Fetching correlated metrics with order volume");

                // Get pod metrics
                var podMetrics = await GetHistoricalPodMetricsAsync(start, end, ct);

                // Group by timestamp
                var grouped = podMetrics
                    .GroupBy(m => m.Timestamp)
                    .Select(g => new PodMetricsWithVolume
                    {
                        Timestamp = g.Key,
                        ReplicaCount = g.First().ReplicaCount,
                        AvgCpuUsage = g.Average(m => m.CpuUsage),
                        AvgMemoryUsage = g.Average(m => m.MemoryUsage),
                        MaxCpuUsage = g.Max(m => m.CpuUsage),
                        MaxMemoryUsage = g.Max(m => m.MemoryUsage),
                        OrderVolume = 0, // Will be filled later
                        ScaledPods = 0,
                        ScalingEvent = "none"
                    })
                    .OrderBy(m => m.Timestamp)
                    .ToList();

                // Add order volume data (from your repository)
                var orderVolumes = await GetOrderVolumesForTimeRangeAsync(start, end, ct);
                
                // Merge data
                foreach (var item in grouped)
                {
                    var matchingVolume = orderVolumes
                        .OrderBy(v => Math.Abs((v.Timestamp - item.Timestamp).TotalMinutes))
                        .FirstOrDefault();

                    if (matchingVolume != null)
                    {
                        item.OrderVolume = matchingVolume.Volume;
                    }
                }

                // Detect scaling events
                for (int i = 1; i < grouped.Count; i++)
                {
                    var current = grouped[i];
                    var previous = grouped[i - 1];

                    if (current.ReplicaCount != previous.ReplicaCount)
                    {
                        current.ScalingEvent = current.ReplicaCount > previous.ReplicaCount 
                            ? "scale_up" 
                            : "scale_down";
                        current.ScaledPods = Math.Abs(current.ReplicaCount - previous.ReplicaCount);
                    }
                }

                _logger.LogInformation($"✅ Retrieved {grouped.Count} correlated metrics entries");
                return grouped;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to get correlated metrics");
                throw;
            }
        }

        /// <summary>
        /// Get deployment metrics summary
        /// </summary>
        public async Task<DeploymentMetricsSummary> GetDeploymentMetricsSummaryAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            try
            {
                var correlatedData = await GetCorrelatedMetricsAsync(start, end, ct);

                var summary = new DeploymentMetricsSummary
                {
                    StartTime = start,
                    EndTime = end,
                    TotalSamples = correlatedData.Count,
                    AvgReplicas = (int)correlatedData.Average(d => d.ReplicaCount),
                    MaxReplicas = correlatedData.Max(d => d.ReplicaCount),
                    MinReplicas = correlatedData.Min(d => d.ReplicaCount),
                    AvgCpuUtilization = correlatedData.Average(d => d.AvgCpuUsage),
                    MaxCpuUtilization = correlatedData.Max(d => d.MaxCpuUsage),
                    AvgMemoryUtilization = correlatedData.Average(d => d.AvgMemoryUsage),
                    MaxMemoryUtilization = correlatedData.Max(d => d.MaxMemoryUsage),
                    HighTrafficEvents = correlatedData.Count(d => d.OrderVolume > 7000),
                    ScalingEvents = correlatedData
                        .Where(d => d.ScalingEvent != "none")
                        .Select(d => new ScalingEvent
                        {
                            Timestamp = d.Timestamp,
                            Type = d.ScalingEvent,
                            FromReplicas = d.ReplicaCount - d.ScaledPods,
                            ToReplicas = d.ReplicaCount,
                            Reason = d.OrderVolume > 7000 ? "high_traffic" : "low_traffic"
                        })
                        .ToList()
                };

                return summary;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Failed to get deployment metrics summary");
                throw;
            }
        }

        /// <summary>
        /// Analyze correlation between order volume and resource usage
        /// </summary>
        public async Task<CorrelationAnalysis> AnalyzeCorrelationAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            var metrics = await GetHistoricalMetricsAsync(start, end, ct);

            if (!metrics.Any())
                return null;

            var analysis = new CorrelationAnalysis
            {
                StartTime = start,
                EndTime = end,
                TotalRecords = metrics.Count
            };

            // Calculate correlation between order volume and CPU
            var volumeCpuCorrelation = CalculateCorrelation(
                metrics.Select(m => (double)m.OrderVolume).ToArray(),
                metrics.Select(m => m.AvgCpuUsage).ToArray()
            );
            analysis.VolumeCpuCorrelation = volumeCpuCorrelation;

            // Calculate correlation between order volume and Memory
            var volumeMemoryCorrelation = CalculateCorrelation(
                metrics.Select(m => (double)m.OrderVolume).ToArray(),
                metrics.Select(m => m.AvgMemoryUsage).ToArray()
            );
            analysis.VolumeMemoryCorrelation = volumeMemoryCorrelation;

            // Calculate correlation between order volume and Replicas
            var volumeReplicasCorrelation = CalculateCorrelation(
                metrics.Select(m => (double)m.OrderVolume).ToArray(),
                metrics.Select(m => (double)m.ReplicaCount).ToArray()
            );
            analysis.VolumeReplicasCorrelation = volumeReplicasCorrelation;

            // Find optimal resource usage patterns
            analysis.OptimalCpuPerOrder = metrics
                .Where(m => m.OrderVolume > 0)
                .Select(m => m.AvgCpuUsage / m.OrderVolume)
                .Average();

            analysis.OptimalMemoryPerOrder = metrics
                .Where(m => m.OrderVolume > 0)
                .Select(m => m.AvgMemoryUsage / m.OrderVolume)
                .Average();

            // Find high traffic events with low resource utilization (opportunities for optimization)
            analysis.Opportunities = metrics
                .Where(m => m.OrderVolume > 5000 && m.AvgCpuUsage < 0.3)
                .Select(m => new OptimizationOpportunity
                {
                    Timestamp = m.Timestamp,
                    OrderVolume = m.OrderVolume,
                    CurrentCpuUsage = m.AvgCpuUsage,
                    CurrentMemoryUsage = m.AvgMemoryUsage,
                    CurrentReplicas = m.ReplicaCount,
                    SuggestedReplicas = Math.Max(2, (int)Math.Ceiling(m.OrderVolume / 1000.0)),
                    Reason = "Low CPU utilization during high traffic"
                })
                .ToList();

            return analysis;
        }

        /// <summary>
        /// Calculate Pearson correlation coefficient
        /// </summary>
        private double CalculateCorrelation(double[] x, double[] y)
        {
            if (x.Length != y.Length || x.Length < 2)
                return 0;

            var n = x.Length;
            var sumX = x.Sum();
            var sumY = y.Sum();
            var sumXY = x.Zip(y, (xi, yi) => xi * yi).Sum();
            var sumX2 = x.Select(xi => xi * xi).Sum();
            var sumY2 = y.Select(yi => yi * yi).Sum();

            var numerator = n * sumXY - sumX * sumY;
            var denominator = Math.Sqrt((n * sumX2 - sumX * sumX) * (n * sumY2 - sumY * sumY));

            if (denominator == 0)
                return 0;

            return numerator / denominator;
        }
    }

public class ApplicationDbContext : DbContext
{
    public ApplicationDbContext(DbContextOptions<ApplicationDbContext> options)
        : base(options)
    {
    }

    public DbSet<Order> Orders { get; set; }
    public DbSet<OrderVolumeHistory> OrderVolumeHistory { get; set; }
    public DbSet<HistoricalMetricsRecord> HistoricalMetrics { get; set; }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // Orders index
        modelBuilder.Entity<Order>()
            .HasIndex(o => o.CreatedAt)
            .HasDatabaseName("IX_Orders_CreatedAt");

        // Historical metrics indexes for efficient querying
        modelBuilder.Entity<HistoricalMetricsRecord>()
            .HasIndex(h => h.Timestamp)
            .HasDatabaseName("IX_HistoricalMetrics_Timestamp");

        modelBuilder.Entity<HistoricalMetricsRecord>()
            .HasIndex(h => new { h.Timestamp, h.OrderVolume })
            .HasDatabaseName("IX_HistoricalMetrics_Timestamp_Volume");

        // Order volume history index

                {

                    var isHighTraffic = currentVolume > _config.HighTrafficThreshold;

        /// <summary>
        /// Store multiple historical metrics records
        /// </summary>
        public async Task StoreHistoricalMetricsAsync(
            List<HistoricalMetricsRecord> records,
            CancellationToken ct = default)
        {
            if (records == null || !records.Any())
                return;

            await _semaphore.WaitAsync(ct);
            try
            {
                // Use bulk insert for better performance
                await _context.HistoricalMetrics.AddRangeAsync(records, ct);
                await _context.SaveChangesAsync(ct);

                _logger.LogInformation($"✅ Stored {records.Count} historical metrics records");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to store historical metrics");
                throw;
            }
            finally
            {
                _semaphore.Release();
            }
        }

        /// <summary>
        /// Store pod metrics for each individual pod
        /// </summary>
        public async Task StorePodMetricsAsync(
            List<PodMetricRecord> podMetrics,

            await _semaphore.WaitAsync(ct);

                _semaphore.Release();

                    .Where(s => s.Timestamp < cutoffDate)
                    .ToListAsync(ct);

                {
                    _context.ScalingEvents.RemoveRange(oldEvents);

                }

                await _context.SaveChangesAsync(ct);

                _logger.LogInformation($"✅ Cleanup complete. Retention: {retentionDays} days");

                _logger.LogError(ex, "❌ Failed to cleanup old data");

                // 2. Get detailed pod metrics
                var podMetrics = await GetDetailedPodMetricsAsync(start.Value, end.Value, ct);

                // 3. Get scaling events
                var scalingEvents = await GetScalingEventsAsync(start.Value, end.Value, ct);

                // 4. Generate and store deployment summary
                await GenerateAndStoreSummaryAsync(start.Value, end.Value, historicalMetrics, ct);

                // 5. Cleanup old data
                await _storageService.CleanupOldDataAsync(30, ct);

                _logger.LogInformation($"✅ All metrics collected and stored successfully");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to collect and store all metrics");
                throw;
            }
        }

        /// <summary>
        /// Get historical metrics with automatic storage
        /// </summary>
        public async Task<List<HistoricalMetricsRecord>> GetHistoricalMetricsWithStorageAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            try
            {
                // Get raw metrics from Prometheus
                var rawMetrics = await GetRawHistoricalMetricsAsync(start, end, ct);

                // Convert to historical records
                var records = new List<HistoricalMetricsRecord>();
                var orderVolumes = await GetOrderVolumesForTimeRangeAsync(start, end, ct);

                foreach (var metric in rawMetrics)
                {
                    var orderVolume = orderVolumes
                        .Where(v => v.Timestamp >= metric.Timestamp.AddMinutes(-5) &&
                                   v.Timestamp <= metric.Timestamp.AddMinutes(5))
                        .OrderBy(v => Math.Abs((v.Timestamp - metric.Timestamp).TotalSeconds))
                        .FirstOrDefault();

                    var record = new HistoricalMetricsRecord
                    {
                        Timestamp = metric.Timestamp,
                        ReplicaCount = metric.ReplicaCount,
                        AvgCpuUsage = metric.AvgCpuUsage,
                        AvgMemoryUsage = metric.AvgMemoryUsage,
                        MaxCpuUsage = metric.MaxCpuUsage,
                        MaxMemoryUsage = metric.MaxMemoryUsage,
                        MinCpuUsage = metric.MinCpuUsage,
                        MinMemoryUsage = metric.MinMemoryUsage,
                        OrderVolume = orderVolume?.Volume ?? 0,
                        CumulativeOrders = orderVolume?.CumulativeOrders ?? 0,
                        IsHighTraffic = (orderVolume?.Volume ?? 0) > 7000,
                        TrafficMode = (orderVolume?.Volume ?? 0) > 7000 ? "high" : "normal",
                        CpuUtilizationPercentage = metric.AvgCpuUsage / (metric.CpuLimit ?? 1) * 100,
                        MemoryUtilizationPercentage = metric.AvgMemoryUsage / (metric.MemoryLimit ?? 1) * 100
                    };

                    records.Add(record);
                }

                // Store in database
                if (records.Any())
                {
                    await _storageService.StoreHistoricalMetricsAsync(records, ct);
                }

                _logger.LogInformation($"✅ Retrieved and stored {records.Count} historical metrics");
                return records;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to get historical metrics with storage");
                throw;
            }
        }

        /// <summary>
        /// Get detailed pod metrics with storage
        /// </summary>
        public async Task<List<PodMetricRecord>> GetDetailedPodMetricsAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct = default)
        {
            try
            {
                var podRecords = new List<PodMetricRecord>();

                // Query Prometheus for pod-level metrics
                var podCpuQuery = $@"
                    sum(
                        rate(
                            container_cpu_usage_seconds_total{{
                                namespace=""{_namespace}"",
                                pod=~""{_deploymentName}.*"",
                                container!=""
                            }}[5m]
                        )
                    ) by (pod, container)";

                var podMemoryQuery = $@"
                    sum(
                        container_memory_working_set_bytes{{
                            namespace=""{_namespace}"",
                            pod=~""{_deploymentName}.*"",
                            container!=""
                        }}
                    ) by (pod, container)";

                // Get pod details
                var pods = await GetPodsAsync(ct);

                foreach (var pod in pods)
                {
                    var podName = pod.Metadata.Name;

                    // Get CPU for this pod
                    var cpuResult = await QueryRangeAsync(
                        $"{podCpuQuery} and pod=\"{podName}\"",
                        start,
                        end,
                        "5m",
                        ct);

                    // Get Memory for this pod
                    var memoryResult = await QueryRangeAsync(
                        $"{podMemoryQuery} and pod=\"{podName}\"",
                        start,
                        end,
                        "5m",
                        ct);

                    // Parse and store metrics
                    foreach (var timestamp in GetTimestamps(cpuResult, memoryResult))
                    {
                        var cpuValue = GetValueAtTimestamp(cpuResult, podName, timestamp);
                        var memoryValue = GetValueAtTimestamp(memoryResult, podName, timestamp);

                        var record = new PodMetricRecord
                        {
                            Timestamp = timestamp,
                            PodName = podName,
                            DeploymentName = _deploymentName,
                            CpuUsage = cpuValue,
                            MemoryUsage = memoryValue,
                            PodStatus = pod.Status.Phase,
                            NodeName = pod.Spec.NodeName,
                            // Extract request/limits from pod spec if available
                            CpuRequest = GetContainerResource(pod, "cpu", "request"),
                            MemoryRequest = GetContainerResource(pod, "memory", "request"),
                            CpuLimit = GetContainerResource(pod, "cpu", "limit"),
                            MemoryLimit = GetContainerResource(pod, "memory", "limit"),
                            CpuUtilization = cpuValue / (GetContainerResource(pod, "cpu", "limit") ?? 1) * 100,
                            MemoryUtilization = memoryValue / (GetContainerResource(pod, "memory", "limit") ?? 1) * 100
                        };

                        podRecords.Add(record);
                    }
                }

                // Store in database
                if (podRecords.Any())
                {
                    await _storageService.StorePodMetricsAsync(podRecords, ct);
                }

                _logger.LogInformation($"✅ Retrieved and stored {podRecords.Count} pod metrics");
                return podRecords;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to get detailed pod metrics");
                throw;
            }
        }

        /// <summary>
        /// Generate and store deployment summary
        /// </summary>
        public async Task GenerateAndStoreSummaryAsync(
            DateTime start,
            DateTime end,
            List<HistoricalMetricsRecord>? metrics = null,
            CancellationToken ct = default)
        {
            try
            {
                if (metrics == null || !metrics.Any())
                {
                    metrics = await _storageService.GetHistoricalMetricsAsync(start, end, ct);
                }

                if (!metrics.Any())
                {
                    _logger.LogWarning("No metrics available for summary generation");
                    return;
                }

                var summary = new DeploymentMetricsSummary
                {
                    WindowStart = start,
                    WindowEnd = end,
                    WindowType = (end - start).TotalHours <= 2 ? "hourly" :
                                (end - start).TotalHours <= 25 ? "daily" : "weekly",
                    AvgReplicas = (int)metrics.Average(m => m.ReplicaCount),
                    MaxReplicas = metrics.Max(m => m.ReplicaCount),
                    MinReplicas = metrics.Min(m => m.ReplicaCount),
                    AvgCpuUsage = metrics.Average(m => m.AvgCpuUsage),
                    MaxCpuUsage = metrics.Max(m => m.MaxCpuUsage),
                    MinCpuUsage = metrics.Min(m => m.MinCpuUsage),
                    AvgMemoryUsage = metrics.Average(m => m.AvgMemoryUsage),
                    MaxMemoryUsage = metrics.Max(m => m.MaxMemoryUsage),
                    MinMemoryUsage = metrics.Min(m => m.MinMemoryUsage),
                    AvgCpuUtilization = metrics.Average(m => m.CpuUtilizationPercentage),
                    AvgMemoryUtilization = metrics.Average(m => m.MemoryUtilizationPercentage),
                    AvgOrderVolume = (int)metrics.Average(m => m.OrderVolume),
                    MaxOrderVolume = metrics.Max(m => m.OrderVolume),
                    MinOrderVolume = metrics.Min(m => m.OrderVolume),
                    TotalOrders = metrics.Sum(m => m.OrderVolume),
                    HighTrafficMinutes = metrics.Count(m => m.IsHighTraffic),
                    LowTrafficMinutes = metrics.Count(m => !m.IsHighTraffic),
                };

                // Calculate correlations
                if (metrics.Count > 1)
                {
                    summary.VolumeCpuCorrelation = CalculateCorrelation(
                        metrics.Select(m => (double)m.OrderVolume).ToArray(),
                        metrics.Select(m => m.AvgCpuUsage).ToArray());

                    summary.VolumeMemoryCorrelation = CalculateCorrelation(
                        metrics.Select(m => (double)m.OrderVolume).ToArray(),
                        metrics.Select(m => m.AvgMemoryUsage).ToArray());

                    summary.VolumeReplicasCorrelation = CalculateCorrelation(
                        metrics.Select(m => (double)m.OrderVolume).ToArray(),
                        metrics.Select(m => (double)m.ReplicaCount).ToArray());
                }

                // Calculate efficiency
                summary.CpuEfficiency = summary.AvgOrderVolume > 0 ?
                    summary.AvgOrderVolume / (summary.AvgCpuUsage * 1000) : 0;

                summary.MemoryEfficiency = summary.AvgOrderVolume > 0 ?
                    (summary.AvgOrderVolume * 1024 * 1024) / summary.AvgMemoryUsage : 0;

                // Get scaling events count
                var scalingEvents = await _storageService.GetScalingEventsAsync(start, end, ct);
                summary.ScalingEventsCount = scalingEvents.Count;

                // Store summary
                await _storageService.StoreDeploymentMetricsSummaryAsync(summary, ct);

                _logger.LogInformation($"✅ Generated and stored deployment summary for {summary.WindowType} window");
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Failed to generate deployment summary");
                throw;
            }
        }

        // Helper methods for Prometheus queries
        private async Task<List<RawMetric>> GetRawHistoricalMetricsAsync(
            DateTime start,
            DateTime end,
            CancellationToken ct)
        {
            var metrics = new List<RawMetric>();

            // Get pod count over time
            var podCountQuery = $@"
                count(
                    kube_pod_info{{
                        namespace=""{_namespace}"",
                        pod=~""{_deploymentName}.*""
                    }}
                )";

            var podCountResult = await QueryRangeAsync(podCountQuery, start, end, "5m", ct);

            // Get CPU usage
            var cpuQuery = $@"
                sum(
                    rate(
                        container_cpu_usage_seconds_total{{
                            namespace=""{_namespace}"",
                            pod=~""{_deploymentName}.*"",
                            container!=""
                        }}[5m]
                    )
                ) by (pod)";

            var cpuResult = await QueryRangeAsync(cpuQuery, start, end, "5m", ct);

            // Get Memory usage
            var memoryQuery = $@"
                sum(
                    container_memory_working_set_bytes{{
                        namespace=""{_namespace}"",
                        pod=~""{_deploymentName}.*"",
                        container!=""
                    }}
                ) by (pod)";

            var memoryResult = await QueryRangeAsync(memoryQuery, start, end, "5m", ct);

            // Combine data
            var timestamps = GetAllTimestamps(cpuResult, memoryResult, podCountResult);

            foreach (var timestamp in timestamps)
            {
                var cpuValues = GetValuesAtTimestamp(cpuResult, timestamp);
                var memoryValues = GetValuesAtTimestamp(memoryResult, timestamp);
                var podCount = GetValueAtTimestamp(podCountResult, timestamp);

                var metric = new RawMetric
                {
                    Timestamp = timestamp,
                    ReplicaCount = (int)podCount,
                    AvgCpuUsage = cpuValues.Any() ? cpuValues.Average() : 0,
                    MaxCpuUsage = cpuValues.Any() ? cpuValues.Max() : 0,
                    MinCpuUsage = cpuValues.Any() ? cpuValues.Min() : 0,
                    AvgMemoryUsage = memoryValues.Any() ? memoryValues.Average() : 0,
                    MaxMemoryUsage = memoryValues.Any() ? memoryValues.Max() : 0,
                    MinMemoryUsage = memoryValues.Any() ? memoryValues.Min() : 0,
                };

                metrics.Add(metric);
            }

            return metrics;
        }

        private double GetContainerResource(V1Pod pod, string resource, string type)
        {
            // Extract container resources from pod spec
            var container = pod.Spec.Containers.FirstOrDefault();
            if (container == null) return 0;

            if (type == "request")
            {
                return container.Resources?.Requests?.GetValueOrDefault(resource)?.Value ?? 0;
            }
            else // limit
            {
                return container.Resources?.Limits?.GetValueOrDefault(resource)?.Value ?? 0;
            }
        }

        private List<DateTime> GetAllTimestamps(params PrometheusQueryResult[] results)
        {
            var timestamps = new HashSet<DateTime>();

            foreach (var result in results)
            {
                if (result?.Data?.Result != null)
                {
                    foreach (var r in result.Data.Result)
                    {
                        foreach (var value in r.Values)
                        {
                            timestamps.Add(DateTimeOffset.FromUnixTimeSeconds((long)value[0]).DateTime);
                        }
                    }
                }
            }

            return timestamps.OrderBy(t => t).ToList();
        }

        private List<double> GetValuesAtTimestamp(PrometheusQueryResult result, DateTime timestamp)
        {
            var values = new List<double>();

            if (result?.Data?.Result != null)
            {
                foreach (var r in result.Data.Result)
                {
                    foreach (var value in r.Values)
                    {
                        if (DateTimeOffset.FromUnixTimeSeconds((long)value[0]).DateTime == timestamp)
                        {
                            values.Add(value[1]);
                        }
                    }
                }
            }

            return values;
        }

        private double GetValueAtTimestamp(PrometheusQueryResult result, DateTime timestamp)
        {
            var values = GetValuesAtTimestamp(result, timestamp);
            return values.Any() ? values.Average() : 0;
        }

        private double GetValueAtTimestamp(PrometheusQueryResult result, string podName, DateTime timestamp)
        {
            if (result?.Data?.Result != null)
            {
                foreach (var r in result.Data.Result)
                {
                    if (r.Metric.GetValueOrDefault("pod") == podName)
                    {
                        foreach (var value in r.Values)
                        {
                            if (DateTimeOffset.FromUnixTimeSeconds((long)value[0]).DateTime == timestamp)
                            {
                                return value[1];
                            }
                        }
                    }
                }
            }
            return 0;
        }
